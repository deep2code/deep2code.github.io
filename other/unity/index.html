Unity 架构

生命周期

// Unity 脚本生命周期
Awake()          // 首次激活时调用一次
OnEnable()      // 每次启用时调用
Start()         // 第一帧前调用一次
Update()        // 每帧调用
FixedUpdate()   // 固定时间间隔调用（物理）
LateUpdate()    // 所有 Update 后调用
OnDisable()     // 每次禁用时调用
OnDestroy()     // 销毁时调用

MonoBehaviour 核心

public class PlayerController : MonoBehaviour
{
    [SerializeField] private float speed = 5f;
    private Rigidbody rb;

    void Awake()
    {
        rb = GetComponent();
    }

    void Update()
    {
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");
        Vector3 movement = new Vector3(horizontal, 0, vertical);
        transform.Translate(movement * speed * Time.deltaTime);
    }

    void FixedUpdate()
    {
        // 物理操作在 FixedUpdate 中
        rb.AddForce(Vector3.up * 9.8f);
    }

    void OnCollisionEnter(Collision collision)
    {
        Debug.Log($"Collision with {collision.gameObject.name}");
    }
}

物理系统

刚体控制

public class PhysicsController : MonoBehaviour
{
    private Rigidbody rb;

    void Start()
    {
        rb = GetComponent();
        rb.mass = 1f;
        rb.drag = 0.5f;
        rb.angularDrag = 0.5f;
        rb.useGravity = true;
        rb.isKinematic = false;
    }

    void ApplyForce()
    {
        // 施加力（物理引擎处理）
        rb.AddForce(Vector3.forward * 10f, ForceMode.Impulse);
    }

    void ApplyVelocity()
    {
        // 直接设置速度
        rb.velocity = Vector3.forward * 5f;
    }

    void MovePosition()
    {
        // 运动学移动（不受物理影响）
        rb.MovePosition(transform.position + Vector3.forward * Time.fixedDeltaTime);
    }
}

资源管理

Addressables 系统

using UnityEngine.AddressableAssets;
using UnityEngine.ResourceManagement.AsyncOperations;

// 异步加载
public async void LoadAsset()
{
    var handle = Addressables.LoadAssetAsync("PlayerPrefab");
    await handle.Task;

    if (handle.Status == AsyncOperationStatus.Succeeded)
    {
        GameObject player = Instantiate(handle.Result);
    }
}

// 加载多个
public async void LoadMultiple()
{
    var handle = Addressables.LoadAssetsAsync("Enemies", null);
    await handle.Task;

    foreach (var enemy in handle.Result)
    {
        Debug.Log(enemy.name);
    }
}

// 释放
Addressables.Release(handle);

AssetBundle

// 加载 AssetBundle
IEnumerator LoadAssetBundle()
{
    string url = "file://" + Application.streamingAssetsPath + "/myassets.unity3d";
    using (WWW www = new WWW(url))
    {
        yield return www;
        AssetBundle bundle = www.assetBundle;

        // 加载资源
        GameObject prefab = bundle.LoadAsset("Player");
        Instantiate(prefab);

        // 卸载
        bundle.Unload(false);
    }
}

DOTS 架构

ECS 基础

using Unity.Entities;
using Unity.Transforms;
using Unity.Mathematics;

// 组件数据（纯数据）
public struct MoveSpeed : IComponentData
{
    public float Value;
}

// 系统（逻辑）
public partial struct MoveSystem : ISystem
{
    public void OnUpdate(ref SystemState state)
    {
        float deltaTime = SystemAPI.Time.DeltaTime;

        foreach (var (transform, speed) in
            SystemAPI.Query, RefRO>())
        {
            transform.ValueRW.Position += new float3(0, 0, 1) * speed.ValueRO.Value * deltaTime;
        }
    }
}

// 组件附加到实体
public class Spawner : MonoBehaviour
{
    public GameObject prefab;

    void Start()
    {
        var world = World.DefaultGameObjectInjectionWorld;
        var entityManager = world.EntityManager;

        var entity = entityManager.CreateEntity();
        entityManager.AddComponentData(entity, new LocalTransform { Position = float3.zero });
        entityManager.AddComponentData(entity, new MoveSpeed { Value = 5f });
    }
}

Burst 编译

using Unity.Burst;
using Unity.Mathematics;

[BurstCompile]
public partial struct MathJob : IJob
{
    public float[] Input;
    public float[] Output;

    public void Execute()
    {
        for (int i = 0; i < Input.Length; i++)
        {
            Output[i] = math.sin(Input[i]) * math.cos(Input[i]);
        }
    }
}

Unity AI

导航系统

using UnityEngine.AI;

public class NavAgentController : MonoBehaviour
{
    private NavMeshAgent agent;

    void Start()
    {
        agent = GetComponent();
        agent.speed = 3.5f;
        agent.angularSpeed = 120f;
        agent.acceleration = 8f;
    }

    void Update()
    {
        if (Input.GetMouseButtonDown(0))
        {
            Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
            if (Physics.Raycast(ray, out RaycastHit hit))
            {
                agent.SetDestination(hit.point);
            }
        }
    }

    void OnTargetReached()
    {
        Debug.Log("Target reached!");
    }
}
