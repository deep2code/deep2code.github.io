MonoBehaviour 核心

public class PlayerController : MonoBehaviour
{
    [SerializeField] private float speed = 5f;
    private Rigidbody rb;

    void Awake()
    {
        rb = GetComponent();
    }

    void Update()
    {
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");
        Vector3 movement = new Vector3(horizontal, 0, vertical);
        transform.Translate(movement * speed * Time.deltaTime);
    }

    void FixedUpdate()
    {
        // 物理操作在 FixedUpdate 中
        rb.AddForce(Vector3.up * 9.8f);
    }

    void OnCollisionEnter(Collision collision)
    {
        Debug.Log($"Collision with {collision.gameObject.name}");
    }
}

public class PhysicsController : MonoBehaviour
{
    private Rigidbody rb;

    void Start()
    {
        rb = GetComponent();
        rb.mass = 1f;
        rb.drag = 0.5f;
        rb.angularDrag = 0.5f;
        rb.useGravity = true;
        rb.isKinematic = false;
    }

    void ApplyForce()
    {
        // 施加力（物理引擎处理）
        rb.AddForce(Vector3.forward * 10f, ForceMode.Impulse);
    }

    void ApplyVelocity()
    {
        // 直接设置速度
        rb.velocity = Vector3.forward * 5f;
    }

    void MovePosition()
    {
        // 运动学移动（不受物理影响）
        rb.MovePosition(transform.position + Vector3.forward * Time.fixedDeltaTime);
    }
}

Addressables 系统

using UnityEngine.AddressableAssets;
using UnityEngine.ResourceManagement.AsyncOperations;

// 异步加载
public async void LoadAsset()
{
    var handle = Addressables.LoadAssetAsync("PlayerPrefab");
    await handle.Task;

    if (handle.Status == AsyncOperationStatus.Succeeded)
    {
        GameObject player = Instantiate(handle.Result);
    }
}

// 加载多个
public async void LoadMultiple()
{
    var handle = Addressables.LoadAssetsAsync("Enemies", null);
    await handle.Task;

    foreach (var enemy in handle.Result)
    {
        Debug.Log(enemy.name);
    }
}

// 释放
Addressables.Release(handle);

    }

Unity 引擎架构深度解析

Unity 核心架构

Unity 是目前最流行的跨平台游戏引擎，理解其内部架构是开发高性能游戏的基础：

  • 实体-组件-系统（ECS 架构思想）：Unity 的核心设计是组件式架构（Component-Based Architecture），虽然不是纯 ECS，但思想一致：
    • GameObject（游戏对象）：场景中的基本实体，本身是空的容器（没有任何功能），通过挂载 Component（组件）获得功能。GameObject 有 Transform（变换，位置/旋转/缩放，每个 GameObject 必有）、名称、标签（Tag）、层（Layer）、激活状态（activeSelf）
    • Component（组件）：附加到 GameObject 上的功能模块，如 MonoBehaviour（自定义脚本）、Renderer（渲染器，MeshRenderer/SkinnedMeshRenderer）、Collider（碰撞器，BoxCollider/SphereCollider）、Rigidbody（刚体，物理）、Camera（相机）、Light（光源）、AudioSource（音频源）。组件可以通过 GetComponent<T>() 获取，通过 AddComponent<T>() 添加
    • Scene（场景）：游戏的一个关卡/界面，包含多个 GameObject 和它们的组件。场景是可序列化的资源（.unity 文件），可以异步加载（SceneManager.LoadSceneAsync）。DontDestroyOnLoad 的对象在场景切换时保留
    • Prefab（预制体）：可复用的 GameObject 模板，包含 GameObject 及其子对象和组件的配置。修改 Prefab 会自动更新所有实例（除非实例覆盖了某些属性）。Prefab Variant（预制体变体）支持继承和覆盖（类似类的继承）。Prefab 是 Unity 工作流的核心（减少重复劳动、统一管理）
  • MonoBehaviour 生命周期：自定义脚本继承 MonoBehaviour，Unity 按固定顺序调用生命周期方法：
    1. Editor 阶段：Reset（编辑器中重置组件时调用）、OnValidate（编辑器中修改属性时调用，可用于数据校验）
    2. 初始化阶段：Awake（对象激活时调用一次，早于 Start，用于初始化自身，此时其他对象可能还未初始化）、OnEnable（对象启用时调用，可能多次调用）、Start（第一帧 Update 之前调用一次，用于初始化依赖其他对象的逻辑，此时所有对象的 Awake 已完成）
    3. 物理阶段：FixedUpdate（固定时间间隔调用，默认 0.02 秒，用于物理计算，与帧率无关）、OnTriggerXXX（触发器事件，Enter/Stay/Exit）、OnCollisionXXX（碰撞事件，Enter/Stay/Exit）
    4. 游戏逻辑阶段：Update（每帧调用一次，用于游戏逻辑，间隔不固定，受帧率影响）、LateUpdate（所有 Update 完成后调用，用于依赖其他对象 Update 结果的逻辑，如相机跟随——确保角色先移动，相机再跟随）
    5. 渲染阶段：OnWillRenderObject、OnPreCull、OnBecameVisible/Invisible、OnPreRender、OnRenderObject、OnPostRender、OnRenderImage（屏幕后处理，OnRenderImage 在 2022+ 被弃用，改用 Render Pipeline Feature）
    6. 销毁阶段：OnDisable（对象禁用时调用）、OnDestroy（对象销毁时调用，用于清理资源）、OnApplicationQuit（应用退出时调用）
    注意：Awake 和 Start 的区别——Awake 在对象实例化时立即调用（即使脚本未启用），Start 在第一帧 Update 前且脚本启用时调用。如果对象初始未激活（SetActive(false)），Awake 不会调用（激活时才调用），Start 同理。OnEnable 在每次启用时调用（可能多次），Awake/Start 只调用一次。
  • 渲染管线（Render Pipeline）：
    • Built-in Render Pipeline（内置渲染管线）：Unity 传统的渲染管线，固定的渲染流程，可通过 OnRenderImage 和 CommandBuffer 扩展。兼容性好（所有资源/插件都支持），但可定制性差，性能不如 SRP
    • URP（Universal Render Pipeline，通用渲染管线）：Unity 官方的跨平台可编程渲染管线，基于 Scriptable Render Pipeline（SRP）框架。URP 优化了性能（单通道前向渲染、SRP Batcher、GPU Instancing），支持移动端/主机/PC/VR 全平台。可通过 Renderer Feature 扩展（如自定义后处理、描边效果）。URP 是新项目的推荐选择
    • HDRP（High Definition Render Pipeline，高清渲染管线）：面向高端平台（PC/PS5/Xbox Series X）的高质量渲染管线，支持光线追踪（Ray Tracing）、高级材质（Layered Lit、Clear Coat、Iridescence）、体积光照（Volumetric Fog）、高级后处理（Bloom、Color Grading、Motion Blur）。HDRP 画质高但性能要求高，不适合移动端
    • SRP Batcher：Unity 的渲染优化技术，通过减少 Draw Call 的 CPU 开销（将材质参数存入 GPU 常量缓冲区，避免每帧设置材质参数的 CPU 开销），大幅提高渲染性能（特别是大量材质不同的对象）。SRP Batcher 要求材质使用 SRP Shader（URP/HDRP 的 Shader），且不使用 MaterialPropertyBlock（会破坏 SRP Batcher）

8大常见陷阱与最佳实践

  1. Update 中执行耗时操作导致帧率下降：Update 每帧调用，如果在 Update 中执行耗时操作（如复杂的数学计算、大量 GameObject 查找、字符串拼接、文件 IO、网络请求、JSON 解析），会导致每帧时间超过 16ms（60fps）或 33ms（30fps），帧率下降（卡顿）。解决：将耗时操作移出 Update：只在需要时执行（如事件驱动，而不是每帧检查）、使用计时器间隔执行（如每 0.5 秒执行一次 AI 决策，而不是每帧）、使用协程（Coroutine）分帧执行（将大任务拆成多帧，如 yield return null 每帧执行一部分）。缓存组件引用：不要在 Update 中调用 GetComponent<T>()、GameObject.Find()、FindObjectOfType()（这些操作耗时，特别是 Find 是 O(n) 遍历所有对象）。在 Awake/Start 中获取并缓存为成员变量（private Rigidbody _rb; void Awake() { _rb = GetComponent<Rigidbody>(); }）。物理计算放在 FixedUpdate：物理相关的操作（Rigidbody 的 AddForce、MovePosition，碰撞检测）放在 FixedUpdate（固定时间间隔，物理引擎稳定），不要放在 Update（帧率波动导致物理不稳定）。使用对象池：频繁创建/销毁 GameObject（如子弹、敌人、特效）会导致 GC 分配和性能开销（Instantiate/Destroy 耗时，且触发 GC）。使用对象池（Object Pooling）：预创建一批对象，需要时取出来激活（SetActive(true)），用完后回收（SetActive(false)），不销毁。Unity 2021+ 内置 ObjectPool 类（UnityEngine.Pool）。字符串优化：避免在 Update 中字符串拼接（+ 操作符会创建新字符串，产生 GC），使用 StringBuilder 或缓存字符串。注意：性能优化的原则是"先测量，再优化"——使用 Unity Profiler（Window > Analysis > Profiler）分析 CPU/GPU/内存瓶颈，找到真正的热点再优化，不要盲目优化
  2. 内存管理不当导致内存泄漏和 GC 卡顿：Unity 使用 Mono/IL2CPP 的自动垃圾回收（GC），但管理不当会导致：内存泄漏（对象被引用但不再使用，GC 无法回收，内存持续增长，最终 OOM 崩溃）、GC 卡顿（GC 回收时暂停所有线程（Stop The World），如果堆很大或引用关系复杂，GC 可能需要几十毫秒，导致明显卡顿）、频繁 GC（每帧都有 GC 分配，导致 GC 频繁触发，帧率不稳定）。解决：减少 GC 分配：避免在 Update/FixedUpdate 中创建引用类型对象（new Class、字符串拼接、装箱（值类型转 object，如 int 传给 object 参数）、params 数组、foreach 遍历（某些情况下会分配枚举器，虽然 Unity 5.6+ 对 List 的 foreach 已优化，但自定义集合可能分配）、LINQ（很多 LINQ 操作会分配闭包和迭代器）。使用值类型（struct）代替引用类型（class）（值类型在栈上分配，不产生 GC，但注意大结构体的复制开销）。对象池：频繁创建/销毁的对象（GameObject、数组、List、StringBuilder）使用对象池复用，避免 GC 分配。内存泄漏排查：使用 Unity Memory Profiler（Package Manager 安装 Memory Profiler 包）分析内存快照（Snapshot），查看哪些对象占用内存、哪些对象被意外引用（导致无法回收）。常见泄漏原因：静态变量引用 GameObject（静态变量不会被 GC，引用的对象也无法回收，场景切换后对象仍在内存中）、事件订阅未取消（someEvent += OnEvent; 但销毁时未 -=，事件发布者持有订阅者引用，订阅者无法回收）、协程未停止（启动的协程持有 MonoBehaviour 引用，对象销毁后协程可能仍在运行）、DontDestroyOnLoad 对象累积。控制 GC 时机：在加载场景或过场动画时主动调用 System.GC.Collect() 触发 GC（此时玩家不关注帧率），避免游戏过程中 GC 卡顿。使用 GC.TryStartNoGCRegion（.NET 4.x）在关键游戏阶段禁止 GC。注意：Unity 的 GC 是分代的（Generational GC），新对象在 Gen 0（回收快），长期存活的对象晋升到 Gen 1/Gen 2（回收慢）。减少长期存活对象的数量可以减少 Gen 2 GC 的开销
  3. Draw Call 过多导致渲染性能瓶颈：Draw Call 是 CPU 向 GPU 发送渲染指令的调用，每个 Draw Call 有 CPU 开销（设置渲染状态、提交几何体）。大量 Draw Call（如几千上万）会导致 CPU 成为瓶颈（GPU 等待 CPU 提交指令，GPU 利用率低，帧率上不去）。特别是移动端，CPU 性能有限，Draw Call 更敏感。解决：静态批处理（Static Batching）：将标记为 Static 的 GameObject（在 Inspector 右上角勾选 Static）的网格合并为一个大网格，一次 Draw Call 渲染所有静态对象。要求：对象是静态的（不移动）、使用相同材质（不同材质会打断批处理）、网格顶点数有限制（静态批处理会合并网格，增加内存占用）。静态批处理在构建时或运行时自动完成。动态批处理（Dynamic Batching）：将使用相同材质的小网格（顶点数 < 300，且使用相同的顶点属性布局）在运行时动态合并为一个 Draw Call。动态批处理适用于小的、移动的对象（如粒子、小道具）。注意：动态批处理有 CPU 开销（每帧合并网格），且限制多（缩放不同的对象不能合并、使用多材质的不能合并、光照贴图不同的不能合并）。URP 中动态批处理默认可能关闭，需要手动开启。GPU Instancing：使用相同网格和相同材质的多个对象，通过一次 Draw Call 渲染多个实例（每个实例有不同的位置/旋转/缩放/颜色，存储在实例缓冲区）。适用于大量重复对象（如树木、草丛、敌人、子弹）。要求：Shader 支持 GPU Instancing（URP/HDRP 的 Lit Shader 默认支持，自定义 Shader 需要添加 #pragma multi_compile_instancing）、使用 MaterialPropertyBlock 设置每实例属性（不要修改材质本身，会破坏批处理）。调用 Graphics.DrawMeshInstanced 或使用 Instancing 渲染器。SRP Batcher：URP/HDRP 中的渲染优化，通过减少材质参数设置的 CPU 开销提高性能。SRP Batcher 不合并 Draw Call（Draw Call 数量不变），但减少每个 Draw Call 的 CPU 时间。要求材质使用 SRP Shader 且不使用 MaterialPropertyBlock。SRP Batcher 对大量不同材质的对象效果好（如建筑场景，每个建筑材质不同，无法批处理，但 SRP Batcher 减少 CPU 开销）。合并材质和纹理：将多个小纹理合并为纹理图集（Texture Atlas），多个材质合并为一个材质（使用纹理图集采样不同区域），减少材质切换（材质切换是 Draw Call 中断的主要原因）。LOD（Level of Detail）：远处的对象使用低多边形网格（减少顶点数和 Draw Call 开销），近处使用高多边形。Unity 内置 LOD Group 组件。遮挡剔除（Occlusion Culling）：不渲染被其他对象遮挡的对象（如墙后面的物体），减少 Draw Call。Unity 内置 Occlusion Culling（Window > Rendering > Occlusion Culling，需要烘焙）。注意：Draw Call 优化的优先级：先看 Profiler 的 Rendering 模块，确认是 CPU 瓶颈（SetPass Call 高、CPU Render 时间长）还是 GPU 瓶颈（GPU 时间长、填充率高）。CPU 瓶颈优化 Draw Call，GPU 瓶颈优化填充率（减少 overdraw、降低分辨率、简化 Shader）
  4. 物理设置不当导致穿透（Tunneling）和性能问题：Unity 的物理引擎（NVIDIA PhysX，3D；Box2D，2D）使用离散碰撞检测（Discrete Collision Detection），默认每帧检测一次碰撞。高速移动的物体（如子弹、快速冲刺的角色）可能在一帧内从碰撞体的一侧移动到另一侧（位移大于碰撞体厚度），导致碰撞检测遗漏（穿透，Tunneling）。物理设置不当还会导致：性能问题（大量 Rigidbody 和 Collider 导致物理计算耗时）、抖动（物体堆叠不稳定、持续微小碰撞）、内存开销（碰撞器过多）。解决：连续碰撞检测（CCD，Continuous Collision Detection）：对高速移动的物体启用 CCD（Rigidbody 的 Collision Detection 设置为 Continuous 或 Continuous Dynamic）。CCD 会在一帧内多次检测碰撞（扫掠检测，Sweep），防止穿透。但 CCD 性能开销大，只对必要的高速物体启用（如子弹），不要所有物体都用 CCD。Continuous（用于高速运动物体，被静态碰撞体碰撞时检测）、Continuous Dynamic（用于高速运动物体，被其他高速/CCD 物体碰撞时检测，开销最大）。限制速度：如果物体速度过快导致穿透，可以限制最大速度（Rigidbody.velocity = Vector3.ClampMagnitude(velocity, maxSpeed)），或增大碰撞体尺寸（让碰撞体更厚，不容易被穿透）。固定时间步长：物理计算在 FixedUpdate 中执行，时间间隔固定（Edit > Project Settings > Time > Fixed Timestep，默认 0.02 秒 = 50 次/秒）。如果游戏帧率低（如 20fps），每帧会执行多次 FixedUpdate（补偿物理时间），可能导致物理计算耗时增加。如果需要更精确的物理（如赛车游戏），可以减小 Fixed Timestep（如 0.01 = 100 次/秒），但物理计算开销翻倍。物理层（Layer）与碰撞矩阵：使用 Layer 区分不同类型的对象（玩家、敌人、子弹、地面、道具），在 Physics Settings 的碰撞矩阵（Layer Collision Matrix）中设置哪些层之间需要碰撞检测（不需要碰撞的层取消勾选，减少物理计算量）。例如子弹和道具不需要碰撞（子弹只和敌人/地面碰撞），取消勾选可以减少碰撞检测对。碰撞器优化：使用简单碰撞器（BoxCollider、SphereCollider、CapsuleCollider）代替 MeshCollider（网格碰撞器，精确但性能开销大，特别是动态物体）。MeshCollider 只用于静态物体（如地形、复杂建筑），且需要勾选 Convex（凸碰撞器）才能与其他 MeshCollider 碰撞。复合碰撞器（Composite Collider，2D）：将多个简单碰撞器合并为一个，减少物理计算。避免频繁修改 Transform：有 Rigidbody 的物体不要直接修改 Transform.position/rotation（会导致物理引擎重新计算，性能开销大，且可能导致碰撞异常）。使用 Rigidbody 的方法：MovePosition/MoveRotation（运动学刚体，Kinematic）、AddForce/AddTorque（动态刚体，受力运动）、velocity/angularVelocity（直接设置速度，谨慎使用）。注意：物理优化的目标是在保证游戏体验的前提下减少物理计算量。使用 Physics Profiler 模块查看物理耗时（Physics.Processing、Physics.Simulate）
  5. 资源管理不当导致加载慢和内存占用高：Unity 的资源（Texture、Mesh、AudioClip、Material、Prefab、Scene）管理不当会导致：加载时间长（场景切换时加载大量资源，黑屏时间长）、内存占用高（所有资源都加载到内存，不释放，移动端 OOM）、资源重复（同一资源被多次加载，内存中有多份副本）、AssetBundle 依赖混乱（依赖资源未加载或重复加载）。解决：Resources 文件夹的使用：Resources 文件夹中的资源会被打包到一个大的资源文件中，通过 Resources.Load 加载。Resources 的缺点：无法增量更新（修改一个资源需要重新打包整个 Resources）、无法按需下载（所有资源都在安装包中）、内存管理不灵活（Resources.UnloadUnusedAssets 才能释放）。建议：只将少量、需要通过代码动态加载的小资源放在 Resources（如配置文件、通用音效），大量资源使用 Addressables 或 AssetBundle。Addressables（可寻址资源系统）：Unity 官方的现代资源管理系统（Package Manager 安装 Addressables 包），是 AssetBundle 的上层封装。Addressables 的优势：按引用自动管理依赖（不需要手动处理 AssetBundle 依赖）、支持异步加载（Addressables.LoadAssetAsync<T>(key)）、支持内存管理（Addressables.Release(handle) 释放引用，引用计数为 0 时自动卸载）、支持远程分发（资源放在 CDN，按需下载，减小安装包体积）、支持增量更新（只下载变化的资源）。Addressables 是新项目的推荐选择。纹理优化：纹理通常是内存占用最大的资源。优化：使用合适的分辨率（不要用 4096x4096 的纹理显示在小物体上，根据物体在屏幕上的大小选择 512/1024/2048）、使用压缩格式（移动端用 ASTC 或 ETC2，PC 用 DXT/BC，压缩后内存减少 4-8 倍）、使用 Mipmaps（远处使用低分辨率 mipmap，减少采样开销和内存带宽，但增加 33% 内存）、关闭非必要纹理的 Mipmaps（UI 纹理、2D 精灵不需要 Mipmaps）、使用纹理图集（Texture Atlas/Sprite Atlas，将多个小纹理合并为一个大纹理，减少 Draw Call 和内存碎片）。音频优化：音频文件格式选择：短音效（UI 点击、射击）用 WAV/ADPCM（解压快，CPU 开销低）、长音频（背景音乐、语音）用 OGG/Vorbis（压缩率高，流式加载，不占内存）。音频加载类型：Decompress On Load（加载时解压到内存，CPU 开销低但内存占用高，适合短音效）、Compressed In Memory（压缩状态存在内存，播放时解压，CPU 开销高但内存占用低，适合中等长度音频）、Streaming（流式从磁盘读取，几乎不占内存，但有磁盘 IO 开销，适合背景音乐）。场景加载优化：使用异步加载（SceneManager.LoadSceneAsync），在加载过程中显示加载进度条。将大场景拆分为多个小场景，使用 Additive 加载（LoadSceneMode.Additive）动态加载/卸载子场景（如开放世界游戏，根据玩家位置加载附近的区域，卸载远处的区域）。注意：资源管理是 Unity 项目中最容易出问题的部分之一，建议从项目初期就规划好资源管理方案（Addressables），不要后期再迁移（成本高）
  6. 协程（Coroutine）使用不当导致逻辑错误和内存泄漏：协程是 Unity 中用于异步执行和分帧任务的机制（基于迭代器 IEnumerator，yield return 暂停执行）。使用不当会导致：协程泄漏（对象销毁后协程仍在运行，引用已销毁的对象导致 MissingReferenceException，或协程持有对象引用导致内存无法回收）、协程重复启动（每次触发条件都启动新协程，多个协程同时运行导致逻辑混乱）、协程停止不及时（对象禁用/销毁时协程未停止，继续执行导致错误）、yield return 类型用错（导致等待时间不正确或不等待）。解决：协程的生命周期：协程依附于启动它的 MonoBehaviour，MonoBehaviour 被销毁（Destroy）时协程自动停止。但 MonoBehaviour 被禁用（SetActive(false) 或 enabled=false）时，协程不会自动停止（会继续运行）！这是常见的陷阱——对象禁用后协程仍在运行，可能访问已禁用的对象导致问题。需要在 OnDisable 中手动停止协程（StopCoroutine）。管理协程引用：启动协程时保存 Coroutine 引用（_myCoroutine = StartCoroutine(MyCoroutine());），需要停止时用引用停止（StopCoroutine(_myCoroutine);）。不要用字符串方法名停止协程（StopCoroutine("MyCoroutine")，性能差且容易写错）。防止重复启动：启动协程前先停止之前的（if (_myCoroutine != null) StopCoroutine(_myCoroutine); _myCoroutine = StartCoroutine(MyCoroutine());），或用标志位判断协程是否在运行。yield return 类型：yield return null;（等待到下一帧继续）、yield return new WaitForSeconds(seconds);（等待指定秒数，受 Time.timeScale 影响——timeScale=0 时不计时）、yield return new WaitForSecondsRealtime(seconds);（等待真实时间，不受 timeScale 影响）、yield return new WaitForFixedUpdate();（等待到下一次 FixedUpdate）、yield return new WaitForEndOfFrame();（等待到帧末，所有渲染完成后）、yield return StartCoroutine(OtherCoroutine());（等待另一个协程完成）、yield return asyncOperation;（等待异步操作完成，如 SceneManager.LoadSceneAsync、AssetBundle.LoadAssetAsync）、yield return www;（等待 WWW/UnityWebRequest 完成）。注意：new WaitForSeconds 每次都会创建新对象（产生 GC），频繁使用的等待时间可以缓存为成员变量（private WaitForSeconds _wait = new WaitForSeconds(0.5f);，复用同一个对象）。协程 vs async/await：Unity 2017+ 支持 C# 的 async/await（基于 Task），可以替代协程。async/await 的优势：可以返回值（Task<T>）、可以用 try-catch 捕获异常（协程的异常处理麻烦）、可以用 CancellationToken 取消。但 async/await 在 Unity 中有一些坑（需要使用 Unity 的 SynchronizationContext，否则回调在其他线程执行无法访问 Unity API；Task 可能产生 GC）。对于简单的分帧和延迟，协程更轻量；对于复杂的异步逻辑（如多个异步操作的组合、错误处理），async/await 更清晰。注意：协程不是多线程——协程在主线程上执行（和 Update 同一个线程），只是分帧执行。如果协程中执行耗时操作（如复杂计算、文件 IO），仍然会阻塞主线程导致卡顿。耗时操作应该放在线程池（Task.Run）或 Unity Job System 中执行
