阿里云 CLI

安装配置

# 安装
curl -sSL https://get.alibabacloud.com | sh

# 或 pip
pip install aliyun-cli

# 配置
aliyun configure --mode KeyPair
# 输入 AccessKeyId, AccessKeySecret, RegionId

# 验证
aliyun ecs DescribeRegions

常用命令

# ECS
aliyun ecs DescribeInstances
aliyun ecs DescribeInstanceStatus

# OSS
aliyun oss ls oss://bucket/
aliyun oss cp file.txt oss://bucket/

# VPC
aliyun vpc DescribeVpcs

# RDS
aliyun rds DescribeDBInstances

自动化脚本

#!/bin/bash
# 批量启动 ECS
for instance_id in $(aliyun ecs DescribeInstances --OutputType=json | jq -r '.Instances.Instance[] | select(.Status == "Stopped") | .InstanceId'); do
    echo "Starting $instance_id"
    aliyun ecs StartInstance --InstanceId $instance_id
done

Python SDK

from aliyunsdkcore.client import AcsClient
from aliyunsdkecs.request.v20140526 import DescribeInstancesRequest

# 初始化客户端
client = AcsClient(
    '',
    '',
    'cn-hangzhou'
)

# 查询 ECS 实例
request = DescribeInstancesRequest.DescribeInstancesRequest()
request.set_PageSize(10)

response = client.do_action_with_exception(request)
print(response.decode('utf-8'))

Terraform 支持

# main.tf
provider "alicloud" {
  region = "cn-hangzhou"
}

resource "alicloud_vpc" "main" {
  name       = "my-vpc"
  cidr_block = "10.0.0.0/16"
}

resource "alicloud_vswitch" "main" {
  vpc_id     = alicloud_vpc.main.id
  cidr_block = "10.0.1.0/24"
  zone_id    = "cn-hangzhou-h"
}

resource "alicloud_instance" "web" {
  instance_name = "web-server"
  image_id      = "centos_7_9_x64_20G_alibase_20231221.vhd"
  instance_type = "ecs.t6.large"
  vswitch_id    = alicloud_vswitch.main.id
}

成本优化

优化策略

  • 预留实例券：降低按量付费成本
  • 竞价实例：成本可降低 90%+
  • 资源组：按项目/部门管理资源
  • 标签策略：按标签分摊成本
  • Serverless：按需付费，无请求不计费

成本工具

  • 费用中心：账单查看与分析
  • 资源编排：成本估算
  • 预算管理：设置消费预警
  • 节省计划：承诺消费换取折扣
