  • 扩展组件

    • 实际上就是 zip 压缩后，直接改后缀名 xpi
    • 注意不能直接压缩整个目录，应该选中扩展文件本身压缩，否则会被判为损坏（多了一层目录）。
    • 现代 Firefox 正式版强制要求扩展通过 AMO 签名，否则不能安装运行

  • "此附件组件无法安装，未通过验证"

    • 打开 Firefox 浏览器
    • 地址栏输入"about:config"
    • 搜索"xpinstall.signatures.required"设置项，双击改为"false"，重启
    • 再把 xpi 拖进 Firefox 窗口便会提示是否安装
    • 注意：xpinstall.signatures.required 仅对 Developer Edition / Nightly 生效；正式版强制 AMO 签名，未签名扩展需经 about:debugging 临时加载

配置优化

about:config 优化

// 性能优化
privacy.trackingprotection.enabled = true      // 跟踪保护
browser.cache.memory.enable = true             // 内存缓存
browser.sessionstore.max_tabs_undo = 20        // 标签页撤销数量

// 界面优化
browser.tabs.closeWindowWithLastTab = false    // 关闭最后一个标签不关闭窗口
browser.tabs.loadInBackground = true           // 中键点击后台打开
toolkit.legacyUserProfileCustomizations.stylesheets = true  // 主题支持

// 网络优化
network.http.max-connections-per-server = 20   // 单服务器连接数
network.http.pipelining = true                 // HTTP 管道化
network.http.pipelining.maxrequests = 8        // 管道化请求数

// 开发者工具
devtools.webconsole.persistlog = true          // 控制台日志持久化
devtools.debugger.new-instrumentation = true   // 新调试器

user.js 配置

// user.js 放在配置目录下
// macOS: ~/Library/Application Support/Firefox/Profiles/xxx.default-release/

// 隐私优化
user_pref("privacy.trackingprotection.enabled", true);
user_pref("privacy.trackingprotection.cryptomining.enabled", true);
user_pref("privacy.trackingprotection.fingerprinting.enabled", true);

// 性能优化
user_pref("browser.tabs.remote.warmup.enabled", false);
user_pref("content.predictor.enabled", false);
user_pref("network.prefetch-next", false);

// 禁用自动更新（可选）
user_pref("app.update.auto", false);
user_pref("app.update.enabled", false);
