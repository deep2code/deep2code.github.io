抓 HTTPS,http2 包

  • 配置 firefox
#设置环境变量,让firefox编入通讯密钥,调试完成后记得删除这个环境变量,它会影响速度和浪费空间
export SSLKEYLOGFILE=~/sslkeylog.log
#命令行启动,让firefox继承环境变量,也防止firefox后台运行,没有真正重启
/Applications/Firefox.app/Contents/MacOS/firefox-bin

  • 配置 wireshark

    • 设置通讯密钥文件位置

    • 编辑 → 首选项 → Protocols → TLS → (Pre)-Master-Secret log filename

  • 配置 http2

    • 需要确保 HTTP2 启用: edit → preferences → protocols → HTTP2

过滤规则

  • IP.addr == 192.168.1.1
  • TCP.port == 80
  • HTTP.request.method == "GET"
  • HTTP.response.code == 200
  • DNS.qry.name == "example.com"

基础使用

捕获选项

# 捕获特定接口
sudo wireshark -i eth0

# 捕获特定端口
sudo wireshark -i eth0 -f "port 80"

# 捕获文件大小限制
sudo wireshark -i eth0 -a filesize:10000

# 环形缓冲区
sudo wireshark -i eth0 -b files:10 -b filesize:10000

http

ip.addr == 192.168.1.1
ip.src == 192.168.1.0/24
ip.dst != 10.0.0.1

# 端口过滤
tcp.port == 443
udp.port == 53
tcp.srcport > 1024

# HTTP 过滤
http.request.method == "GET"
http.response.code == 200
http.host contains "example.com"

# 组合过滤
ip.addr == 192.168.1.1 and tcp.port == 80
http and not dns
(tcp.flags.syn == 1) and (tcp.flags.ack == 0)

命令行工具

TShark

# 捕获并保存
tshark -i eth0 -w capture.pcap

# 读取文件并过滤
tshark -r capture.pcap -Y "http.request.method == GET"

# 提取 HTTP 请求
tshark -r capture.pcap -Y "http.request" -T fields -e http.request.method -e http.request.uri

# 统计
tshark -r capture.pcap -q -z io,phs

Dumpcap

# 捕获到文件
dumpcap -i eth0 -w capture.pcap

# 捕获并按大小分割
dumpcap -i eth0 -b filesize:10000 -w capture.pcap

# 捕获特定时间
dumpcap -i eth0 -a duration:60 -w capture.pcap

Editcap

# 转换格式
editcap -F pcap capture.pcapng capture.pcap

# 提取特定包
editcap -r capture.pcap 1-100.pcap 1-100

# 按时间过滤
editcap -r capture.pcap -A "2024-01-01" -B "2024-01-02" output.pcap

Lua 脚本

基础脚本

-- 解析自定义协议
local myproto = Proto("myproto", "My Protocol")

function myproto.dissector(buffer, pinfo, tree)
    local subtree = tree:add(myproto, buffer(), "My Protocol Data")
    subtree:add(buffer(0, 4), "Header")
    subtree:add(buffer(4), "Payload")
end

local tcp_port = DissectorTable.get("tcp.port")
tcp_port:add(8080, myproto)

HTTP 插件

-- 提取 HTTP 请求
local http_ex = Field.new("http.request")

function hook.packet()
    local req = http_ex()
    if req then
        print("HTTP Request: " .. req())
    end
end

register_hooker("postdissector", hook)
