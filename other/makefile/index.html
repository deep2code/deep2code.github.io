常见编译变量参数

GIT_VERSION ?= $(shell git describe --always --tags --match 'v*' --dirty)
COMMIT     ?= $(shell git rev-parse HEAD)
BRANCH     ?= $(shell git rev-parse --abbrev-ref HEAD)

BUILD_HOST ?= $(shell hostname)

变量不同赋值区别

= 是最基本的赋值
:= 是覆盖之前的值
?= 是如果没有被赋值过就赋予等号后面的值
+= 是添加等号后面的值

= 是递归展开：值里的变量引用到使用时才展开，写起来宽松、但变量互相引用时容易"套娃"出意外；:= 是直接展开：定义时立即展开，适合用于确保变量值在定义时确定。

Makefile 高级模式

.PHONY 声明

声明伪目标，避免文件名冲突：

.PHONY: all clean build test install

all: build test

build:
	go build -o bin/app ./cmd/app

test:
	go test -v ./...

clean:
	rm -rf bin/ coverage.out

常用函数

# 字符串替换
$(subst from,to,text)

# 模式替换
$(patsubst pattern,replacement,text)

# 过滤
$(filter pattern...,text)

# 循环
$(foreach var,list,text)

# 调用外部命令
$(shell command)

# 条件
$(if condition,then-part,else-part)

通配符与模式匹配

# 获取所有 .go 文件
SOURCES := $(wildcard *.go)

# 获取所有 .go 文件（递归）
SOURCES := $(shell find . -name '*.go')

# 模式规则
%.o: %.c
	$(CC) -c $< -o $@

# 静态模式规则
OBJS := foo.o bar.o
$(OBJS): %.o: %.c
	$(CC) -c $< -o $@

调试技巧

打印变量

# 打印变量值
$(info VAR_NAME=$(VAR_NAME))

# 调试模式
$(warning VAR_NAME=$(VAR_NAME))

# 打印所有变量
debug:
	@$(foreach V,$(.VARIABLES),$(info $(V) = $($(V))))

模拟执行

# -n 只打印命令不执行
make -n build

# -d 打印调试信息
make -d clean

# --dry-run 同样是模拟
make --dry-run test

自动变量

  • $@：目标文件名
  • $<：第一个依赖文件
  • $^：所有依赖文件
  • $+：所有依赖文件（保留重复）
  • $*：目标去掉后缀的部分
  • $?：比目标更新的依赖
  • $(@D)：目标所在目录
  • $(@F)：目标文件名（不含目录）

包含与模块化

包含其他 Makefile

# 包含配置
-include config.mk

# 条件包含
ifneq ($(wildcard config.mk),)
    include config.mk
endif

# 包含多个文件
include Makefile.vars
include Makefile.targets

模板化 Makefile

# Makefile.template
PROJECT_NAME := myproject
BUILD_DIR := ./build
SRC_DIR := ./src

# 默认目标
.PHONY: all
all: help

# 帮助信息
.PHONY: help
help:
	@echo "Available targets:"
	@echo "  build    - Build the project"
	@echo "  test     - Run tests"
	@echo "  clean    - Clean build artifacts"
	@echo "  docker   - Build Docker image"

Go 项目 Makefile

Go 常用命令

.PHONY: build test lint run clean docker

BINARY_NAME=myapp
BUILD_DIR=./bin
GO=go
GOFLAGS=-v

build:
	$(GO) build $(GOFLAGS) -o $(BUILD_DIR)/$(BINARY_NAME) ./cmd/$(BINARY_NAME)

test:
	$(GO) test -v -coverprofile=coverage.out ./...

test-race:
	$(GO) test -race -v ./...

lint:
	golangci-lint run ./...

run:
	$(GO) run ./cmd/$(BINARY_NAME)

clean:
	rm -rf $(BUILD_DIR) coverage.out

docker:
	docker build -t $(BINARY_NAME):latest .

# 开发快捷命令
dev:
	$(GO) run ./cmd/$(BINARY_NAME) -dev

tidy:
	$(GO) mod tidy

vet:
	$(GO) vet ./...

deps:
	$(GO) mod download
	$(GO) mod verify

多平台构建

.PHONY: build-all build-linux build-darwin build-windows

build-all: build-linux build-darwin build-windows

build-linux:
	GOOS=linux GOARCH=amd64 go build -o bin/linux/$(BINARY_NAME) ./cmd/$(BINARY_NAME)
	GOOS=linux GOARCH=arm64 go build -o bin/linux-arm64/$(BINARY_NAME) ./cmd/$(BINARY_NAME)

build-darwin:
	GOOS=darwin GOARCH=amd64 go build -o bin/darwin/$(BINARY_NAME) ./cmd/$(BINARY_NAME)
	GOOS=darwin GOARCH=arm64 go build -o bin/darwin-arm64/$(BINARY_NAME) ./cmd/$(BINARY_NAME)

build-windows:
	GOOS=windows GOARCH=amd64 go build -o bin/windows/$(BINARY_NAME).exe ./cmd/$(BINARY_NAME)

现代替代工具

Just - 命令运行器

# 安装
brew install just

# justfile 示例
default:
    @just --list

build:
    go build -o bin/app

test:
    go test -v ./...

deploy:
    docker build -t app .
    docker push registry/app:tag

# 运行
just build
just test
just deploy

Taskfile - Task Runner

# Taskfile.yml
version: '3'

tasks:
  build:
    cmds:
      - go build -o bin/app

  test:
    cmds:
      - go test -v ./...

  dev:
    cmds:
      - air
    sources:
      - "**/*.go"
    methods:
      - go build ./...
