Windows 开发环境更新

  • WSL2：Windows Subsystem for Linux 2，接近原生 Linux 性能。
  • Windows Terminal：支持多标签、GPU 加速。
  • PowerShell 7：跨平台，基于 .NET。
  • WinGet：Windows 包管理器。

WSL2 完整指南

安装 WSL2

# 简单方法（推荐）
wsl --install

# 或者手动启用功能
dism.exe /online /enable-feature /featurename:Microsoft-Windows-Subsystem-Linux /all /norestart
dism.exe /online /enable-feature /featurename:VirtualMachinePlatform /all /norestart

# 重启后安装
wsl --install -d Ubuntu-22.04

常用命令

# 查看已安装的发行版
wsl -l -v

# 启动特定的发行版
wsl -d Ubuntu-22.04

# 关闭所有发行版
wsl --shutdown

# 导出/导入发行版
wsl --export Ubuntu ~/ubuntu.tar
wsl --import Ubuntu ~/ubuntu ~/ubuntu.tar

# 在 Windows 中运行 Linux 命令
wsl ls -la
wsl cat /etc/passwd

运行 GUI 应用

# 安装 GUI 支持
sudo apt update
sudo apt install -y x11-apps

# 设置 DISPLAY 环境变量
export DISPLAY=:0

# 运行 GUI 应用
gedit &
# 或者
nautilus &

Docker in WSL2

# 安装 Docker
sudo apt update
sudo apt install -y docker.io

# 启动 Docker 服务
sudo service docker start

# 当前用户加入 docker 组
sudo usermod -aG docker $USER

# 重新登录后使用
docker run hello-world

WinGet 包管理器

基本使用

# 搜索软件
winget search vscode

# 安装软件
winget install Microsoft.VisualStudioCode

# 查看已安装
winget list

# 升级软件
winget upgrade vscode

# 升级所有
winget upgrade --all

# 卸载软件
winget uninstall vscode

WinGet vs Scoop

  • WinGet：官方推荐，支持签名包，企业级
  • Scoop：用户友好，版本管理，CLI 优先
  • Chocolatey：老牌包管理器，库最全

安装 Scoop

# 先安装 Scoop
iwr -useb get.scoop.sh | iex

# 添加 extras bucket
scoop bucket add extras

# 安装软件
scoop install git
scoop install vscode
scoop install docker

# 搜索
scoop search python

# 卸载
scoop uninstall vscode

PowerShell 7 指南

安装 PowerShell 7

# 使用 WinGet
winget install Microsoft.PowerShell

# 或者使用 winget
winget install PowerShell.Preview

常用模块

# 安装常用模块
Install-Module -Name PowerShellGet -Force
Install-Module -Name PSReadLine -AllowClobber
Install-Module -Name oh-my-posh

# 配置 PSReadLine
Import-Module PSReadLine
Set-PSReadLineOption -PredictionSource History
Set-PSReadLineOption -PredictionViewStyle ListView

# 配置 oh-my-posh
oh-my-posh init pwsh --config "$env:POSH_THEMES_PATH/ paradox.omp.json" | Invoke-Expression

常用别名

# 设置别名
Set-Alias -Name ll -Value Get-ChildItem
Set-Alias -Name grep -Value Select-String

# 函数的别名
function which {
    param($command)
    Get-Command $command -ErrorAction SilentlyContinue | Select-Object -ExpandProperty Source
}

Set-Alias -Name which -Value Get-Command

WSL2 + CUDA

# 在 WSL2 中安装 CUDA
wsl

# 添加 CUDA 仓库
wget https://developer.download.nvidia.com/compute/cuda/repos/wsl-ubuntu/x86_64/cuda-wsl-ubuntu.pin
sudo mv cuda-wsl-ubuntu.pin /etc/apt/preferences.d/cuda-repository-pin-600
wget https://developer.download.nvidia.com/compute/cuda/12.0.0/cuda-repo-wsl-ubuntu-12-0-local_12.0.0-1_amd64.deb
sudo dpkg -i cuda-repo-wsl-ubuntu-12-0-local_12.0.0-1_amd64.deb
sudo apt-key add /var/cuda-repo-wsl-ubuntu-12-0-local/7fa2af80.pub
sudo apt-get update
sudo apt-get install cuda
