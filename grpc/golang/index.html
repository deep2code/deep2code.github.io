一个完整的 gRPC-Go 入门示例：server + client + proto 定义 + nginx 反向代理。代码演示一元 RPC（unary RPC）与拦截器（interceptor）的用法。

server

package main

import (
 "context"
 "flag"
 "fmt"
 "log"
 "net"

 "google.golang.org/grpc"
 "google.golang.org/grpc/metadata"
 "google.golang.org/grpc/peer"

 com "xxx.site/myself/grpc-common"
)

var (
 GitHash     = "unknown"
 CompileTime = "unknown"
 port        = flag.Int("port", 8411, "默认端口")
)

// 定义服务端中间件（一元拦截器）
func middleware(ctx context.Context, req interface{},
 info *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (resp interface{}, err error) {
 log.Printf("middleware ctx:%v", ctx)
 log.Printf("middleware req:%v", req)
 log.Printf("middleware info:%v", info)
 log.Printf("middleware handler:%v", handler)
 resp, err = handler(ctx, req)
 log.Printf("middleware resp:%v", resp)
 log.Printf("middleware err:%v", err)
 return
}

type MathServer struct {
 com.UnimplementedMathServer
}

// 各种上下文信息都通过 ctx 中的 valueCtx 传递进来
// 由不同包获取转换：peer / metadata / ServerTransportStream
func (ms *MathServer) Add(ctx context.Context, req *com.AddReq) (*com.AddRsp, error) {
 log.Printf("ctx:%v, req:%v", ctx, req)
 if client, ok := peer.FromContext(ctx); ok {
  log.Printf("client:%v", client)
 }

 if ic, ok := metadata.FromIncomingContext(ctx); ok {
  log.Printf("ic:%v", ic)
 }

 if oc, ok := metadata.FromOutgoingContext(ctx); ok {
  log.Printf("oc:%v", oc)
 }

 sts := grpc.ServerTransportStreamFromContext(ctx)
 log.Printf("sts:%v", sts)

 return &com.AddRsp{
  Result: req.Left + req.Right,
 }, nil
}

func main() {
 flag.Parse()

 // ct, err := credentials.NewServerTLSFromFile(
 //  "grpc.xxx.site.pem",
 //  "grpc.xxx.site.key",
 // )
 // if err != nil {
 //  log.Fatalf("tls file;%v", err)
 // }

 // "tcp" 监听所有可用地址族（IPv4/IPv6 双栈）；只要 IPv4 可写 "tcp4"，强制 IPv6 写 "tcp6"
 l, err := net.Listen("tcp", fmt.Sprintf(":%d", *port))
 if err != nil {
  log.Fatalf("fail listen tcp %d", *port)
 }

 // 安装中间件（默认不启用 TLS）
 s := grpc.NewServer(
  // 增加证书认证
  //grpc.Creds(ct),
  grpc.UnaryInterceptor(middleware),
 )

 com.RegisterMathServer(s, &MathServer{})
 log.Printf(
  "githash:%v, compile:%v,listen:%v",
  GitHash,
  CompileTime,
  l.Addr())

 err = s.Serve(l)
 if err != nil {
  log.Fatalf("fail server:%v", err)
 }
}

2026 提示：一元拦截器是服务端扩展点（鉴权、限流、日志）。更细粒度可用 grpc.UnaryChainInterceptor 链式组合，或借助 OpenTelemetry 的 gRPC 插桩完成链路追踪，替代手写日志中间件。

client

package main

import (
 "context"
 "flag"
 "log"
 "time"

 "google.golang.org/grpc"
 "google.golang.org/grpc/credentials/insecure"

 com "xxx.site/myself/grpc-common"
)

var (
 GitHash     = "unknown"
 CompileTime = "unknown"
 // addr = flag.String("addr", "dns:///grpc.xxx.site", "默认服务端端口")
 addr = flag.String("addr", "127.0.0.1:8411", "默认服务端端口")
)

// 定义客户端中间件（一元拦截器）
func middleware(ctx context.Context, method string,
 req, reply interface{}, c *grpc.ClientConn, invoker grpc.UnaryInvoker, opts ...grpc.CallOption) error {

 log.Printf("middleware ctx:%v", ctx)
 log.Printf("middleware method:%v", method)
 log.Printf("middleware req:%v", req)
 log.Printf("middleware reply:%v", reply)
 log.Printf("middleware conn:%v", c)
 log.Printf("middleware invoker:%v", invoker)
 for pos, opt := range opts {
  log.Printf("middleware pos:%v, opt:%v", pos, opt)
 }

 err := invoker(ctx, method, req, reply, c, opts...)

 log.Printf("middleware err:%v", err)

 return err

}

func main() {
 flag.Parse()

 // ct, err := credentials.NewClientTLSFromFile("grpc.xxx.site.pem", "grpc.xxx.site")
 // if err != nil {
 //  log.Fatalf("err:%v", err)
 // }

 // 新版 gRPC-Go 推荐 grpc.NewClient；grpc.Dial 自 v1.63 起标记废弃
 conn, err := grpc.NewClient(
  *addr,
  // 禁用安全传输（无 TLS）
  grpc.WithTransportCredentials(insecure.NewCredentials()),
  // 采用证书认证（注意跨平台证书兼容性）
  // grpc.WithTransportCredentials(ct),
  // 安装客户端中间件
  grpc.WithUnaryInterceptor(middleware),
 )
 if err != nil {
  log.Fatalf("dial:%v, %v", *addr, err)
 }
 defer conn.Close()

 client := com.NewMathClient(conn)

 req := com.AddReq{
  Left:  10,
  Right: 20,
 }

 resp, err := client.Add(context.Background(), &req)
 if err != nil {
  log.Fatalf("add fail:%v", err)
 }

 log.Printf("resp:%v", resp.Result)
}

为什么改用 grpc.NewClient：grpc.Dial（grpc.WithTimeout 等旧选项）自 gRPC-Go v1.63（2024 年）起标记废弃，官方推荐 grpc.NewClient——它在创建时不再立即阻塞建连，超时与连接管理交给 ConnectParams/拨号选项与 context 处理，语义更清晰，2026 年的 v1.8x 仍保持向后兼容。

nginx 反向代理

gRPC 走 HTTP/2，nginx 1.13.10+ 原生支持 grpc_pass 反代；TLS 场景使用 grpcs:// 前缀：

server_names_hash_bucket_size 64;
server {
    listen 443 ssl http2;
    server_name grpc.xxx.site;

    ssl_protocols TLSv1.2 TLSv1.3;  # TLSv1/1.1 已被 IETF 弃用（RFC 8996，2021-03），2026 年生产环境不应再开启
    ssl_certificate grpc.xxx.site.pem;
    ssl_certificate_key grpc.xxx.site.key;

    ssl_session_cache shared:SSL:1m;
    ssl_session_timeout 5m;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;

    location / {
        # grpc_pass localhost:9000; 等价于 grpc://127.0.0.1:8411 等价于 [::]:8411
        # 使用 gRPC over TLS 时带上 grpcs:
        grpc_pass grpcs://[xxx]:8411;
        client_max_body_size 200M;
    }
}

2026 更新

现代 gRPC Go 项目建议用 buf 管理 proto 文件（lint + breaking + 生成），详见 gRPC 主页。Connect-RPC 是更简单的 gRPC 兼容替代方案，可直接暴露 HTTP/JSON 端点；gRPC-Go 1.8x 的完整版本与安全注意事项同样见主页「gRPC 2026 生态更新」。

Go 1.26 语言与运行时更新

gRPC-Go v1.81 起要求 Go 1.25+，而 Go 1.26.0 已于 2026 年 2 月 10 日发布（当前稳定版 1.26.x，例如 go1.26.6），升级 Go 后请注意以下与本示例直接相关的变化：

  • new(expr)：内置 new 支持表达式参数（如 p := new(42)），生成指向表达式值的指针，简化 JSON/Protobuf 可选字段的指针初始化。
  • errors.AsType：泛型化错误类型断言 errors.AsType[*AppError](err)，编译期类型安全、无反射开销，替代 errors.As。
  • Green Tea GC 默认启用：以 8KiB span 为单元批量标记、并行扫描，小对象场景 GC 开销平均下降 10%~40%，可用 GOEXPERIMENT=nogreenteagc 回退（将在 1.27 移除）。
  • 后量子加密默认化：新增 crypto/hpke（RFC 9180）与 crypto/mlkem 包，TLS 默认启用 X25519MLKEM768 等后量子混合密钥交换。
  • 可观测性增强：runtime/metrics 新增 goroutine/线程调度器指标，配合 goroutineleak 性能分析可在线排查 goroutine 泄漏。
