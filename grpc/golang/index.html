一个完整的 gRPC-Go 入门示例：server + client + proto 定义 + Nginx 反向代理。代码演示一元 RPC（unary RPC）和拦截器（interceptor）的用法。

server

package main

import (
 "context"
 "flag"
 "fmt"
 "log"
 "net"

 "google.golang.org/grpc"
 "google.golang.org/grpc/metadata"
 "google.golang.org/grpc/peer"

 com "xxx.site/myself/grpc-common"
)

var (
 GitHash     = "unknown"
 CompileTime = "unknown"
 port        = flag.Int("port", 8411, "默认端口")
)

// 定义服务端中间件（一元拦截器）
func middleware(ctx context.Context, req interface{},
 info *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (resp interface{}, err error) {
 log.Printf("middleware ctx:%v", ctx)
 log.Printf("middleware req:%v", req)
 log.Printf("middleware info:%v", info)
 log.Printf("middleware handler:%v", handler)
 resp, err = handler(ctx, req)
 log.Printf("middleware resp:%v", resp)
 log.Printf("middleware err:%v", err)
 return
}

type MathServer struct {
 com.UnimplementedMathServer
}

// 各种上下文信息都通过 ctx 中的 valueCtx 传递进来
// 由不同包获取转换：peer / metadata / ServerTransportStream
func (ms *MathServer) Add(ctx context.Context, req *com.AddReq) (*com.AddRsp, error) {
 log.Printf("ctx:%v, req:%v", ctx, req)
 if client, ok := peer.FromContext(ctx); ok {
  log.Printf("client:%v", client)
 }

 if ic, ok := metadata.FromIncomingContext(ctx); ok {
  log.Printf("ic:%v", ic)
 }

 if oc, ok := metadata.FromOutgoingContext(ctx); ok {
  log.Printf("oc:%v", oc)
 }

 sts := grpc.ServerTransportStreamFromContext(ctx)
 log.Printf("sts:%v", sts)

 return &com.AddRsp{
  Result: req.Left + req.Right,
 }, nil
}

func main() {
 flag.Parse()

 // ct, err := credentials.NewServerTLSFromFile(
 //  "grpc.xxx.site.pem",
 //  "grpc.xxx.site.key",
 // )
 // if err != nil {
 //  log.Fatalf("tls file;%v", err)
 // }

 // "tcp" 监听所有可用地址族（IPv4/IPv6 双栈）；只要 IPv4 可写 "tcp4"，强制 IPv6 写 "tcp6"
 l, err := net.Listen("tcp", fmt.Sprintf(":%d", *port))
 if err != nil {
  log.Fatalf("fail listen tcp %d", *port)
 }

 // 安装中间件（默认不启用 TLS）
 s := grpc.NewServer(
  // 增加证书认证
  //grpc.Creds(ct),
  grpc.UnaryInterceptor(middleware),
 )

 com.RegisterMathServer(s, &MathServer{})
 log.Printf(
  "githash:%v, compile:%v,listen:%v",
  GitHash,
  CompileTime,
  l.Addr())

 err = s.Serve(l)
 if err != nil {
  log.Fatalf("fail server:%v", err)
 }
}

2026 提示：一元拦截器是服务端的扩展点（鉴权、限流、日志）。更细粒度可以用 gRPC.UnaryChainInterceptor 链式组合，或借助 OpenTelemetry 的 gRPC 插桩完成链路追踪，替代手写日志中间件。

package main

import (
 "context"
 "flag"
 "log"
 "time"

 "google.golang.org/grpc"
 "google.golang.org/grpc/credentials/insecure"

 com "xxx.site/myself/grpc-common"
)

var (
 GitHash     = "unknown"
 CompileTime = "unknown"
 // addr = flag.String("addr", "dns:///grpc.xxx.site", "默认服务端端口")
 addr = flag.String("addr", "127.0.0.1:8411", "默认服务端端口")
)

// 定义客户端中间件（一元拦截器）
func middleware(ctx context.Context, method string,
 req, reply interface{}, c *grpc.ClientConn, invoker grpc.UnaryInvoker, opts ...grpc.CallOption) error {

 log.Printf("middleware ctx:%v", ctx)
 log.Printf("middleware method:%v", method)
 log.Printf("middleware req:%v", req)
 log.Printf("middleware reply:%v", reply)
 log.Printf("middleware conn:%v", c)
 log.Printf("middleware invoker:%v", invoker)
 for pos, opt := range opts {
  log.Printf("middleware pos:%v, opt:%v", pos, opt)
 }

 err := invoker(ctx, method, req, reply, c, opts...)

 log.Printf("middleware err:%v", err)

 return err

}

func main() {
 flag.Parse()

 // ct, err := credentials.NewClientTLSFromFile("grpc.xxx.site.pem", "grpc.xxx.site")
 // if err != nil {
 //  log.Fatalf("err:%v", err)
 // }

 // 新版 gRPC-Go 推荐 grpc.NewClient；grpc.Dial 自 v1.63 起标记废弃
 conn, err := grpc.NewClient(
  *addr,
  // 禁用安全传输（无 TLS）
  grpc.WithTransportCredentials(insecure.NewCredentials()),
  // 采用证书认证（注意跨平台证书兼容性）
  // grpc.WithTransportCredentials(ct),
  // 安装客户端中间件
  grpc.WithUnaryInterceptor(middleware),
 )
 if err != nil {
  log.Fatalf("dial:%v, %v", *addr, err)
 }
 defer conn.Close()

 client := com.NewMathClient(conn)

 req := com.AddReq{
  Left:  10,
  Right: 20,
 }

 resp, err := client.Add(context.Background(), &req)
 if err != nil {
  log.Fatalf("add fail:%v", err)
 }

 log.Printf("resp:%v", resp.Result)
}

为什么改用 gRPC.NewClient：gRPC.Dial （gRPC.WithTimeout 等旧选项）自 gRPC-Go v1.63（2024 年）起标记废弃，官方推荐 gRPC.NewClient ——它在创建时不再立即阻塞建连，超时和连接管理交给 ConnectParams / 拨号选项和 context 处理，语义更清晰，2026 年的 v1.8x 仍保持向后兼容。

Nginx 反向代理

gRPC 走 HTTP/2，Nginx 1.13.10+ 原生支持 grpc_pass 反代；TLS 场景用 grpcs:// 前缀：

server_names_hash_bucket_size 64;
server {
    listen 443 ssl http2;
    server_name grpc.xxx.site;

    ssl_protocols TLSv1.2 TLSv1.3;  # TLSv1/1.1 已被 IETF 弃用（RFC 8996，2021-03），2026 年生产环境不应再开启
    ssl_certificate grpc.xxx.site.pem;
    ssl_certificate_key grpc.xxx.site.key;

    ssl_session_cache shared:SSL:1m;
    ssl_session_timeout 5m;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;

    location / {
        # grpc_pass localhost:9000; 等价于 grpc://127.0.0.1:8411 等价于 [::]:8411
        # 使用 gRPC over TLS 时带上 grpcs:
        grpc_pass grpcs://[xxx]:8411;
        client_max_body_size 200M;
    }
}

2026 更新

现代 gRPC Go 项目建议用 buf 管理 proto 文件（lint + breaking + 生成），详见 gRPC 主页。Connect-RPC 是更简单的 gRPC 兼容替代方案，可以直接暴露 HTTP/JSON 端点；gRPC-Go 1.8x 的完整版本和安全注意事项同样见主页「gRPC 2026 生态更新」。

Go 1.26 语言与运行时更新

gRPC-Go v1.81 起要求 Go 1.25+，而 Go 1.26.0 已于 2026 年 2 月 10 日发布（当前稳定版 1.26.x，比如 go1.26.6），升级 Go 后注意以下和本示例直接相关的变化：

  • new(expr)：内置 new 支持表达式参数（比如 p := new(42)），生成指向表达式值的指针，简化 JSON / Protobuf 可选字段的指针初始化。
  • errors.AsType：泛型化错误类型断言 errors.AsType[*AppError](err)，编译期类型安全、无反射开销，替代 errors.As。
  • Green Tea GC 默认启用：以 8KiB span 为单元批量标记、并行扫描，小对象场景 GC 开销平均下降 10%~40%，可以用 GOEXPERIMENT=nogreenteagc 回退（将在 1.27 移除）。
  • 后量子加密默认化：新增 crypto/hpke（RFC 9180）和 crypto/mlkem 包，TLS 默认启用 X25519MLKEM768 等后量子混合密钥交换。
  • 可观测性增强：runtime/metrics 新增 goroutine / 线程调度器指标，配合 goroutineleak 性能分析可以在线排查 goroutine 泄漏。

连接管理与 HTTP/2 传输

// gRPC-Go 的连接管理架构：

// ClientConn：客户端连接，管理到一个或多个后端的连接
type ClientConn struct {
    mu           sync.RWMutex
    resolver     resolver.ClientConn  // 服务发现
    balancer     balancer.ClientConn   // 负载均衡
    transport    transport.ClientTransport // HTTP/2 传输
    csMgr        *connectivityStateManager // 连接状态管理
    authority    string
    dopts        dialOptions
}

// 传输层（transport）负责 HTTP/2 帧的读写
// http2Client 实现：
type http2Client struct {
    conn        net.Conn
    readerDone  chan struct{}
    controlBuf  *controlBuffer  // 控制帧缓冲区
    framer      *framer         // 帧读写器
    activeStreamCount int
    maxStreamCount    int32
    initialWindowSize int32
    connectionWindow  *window
    mu          sync.Mutex
    streams     map[uint32]*Stream
}

// 关键设计：
// 1. 一个 TCP 连接上多路复用多个 gRPC 调用（Stream）
// 2. 控制帧（SETTINGS/PING/WINDOW_UPDATE）和数据帧分离处理
// 3. 流控（Flow Control）基于 HTTP/2 WINDOW_UPDATE 帧
// 4. 连接失败时自动重连，指数退避（1s, 2s, 4s, 8s...最大 120s）

// 流（Stream）的生命周期：
type Stream struct {
    id           uint32
    method       string
    sendQuota    int32         // 发送配额（流控）
    recvQuota    int32         // 接收配额
    headerChan   chan struct{} // 头部到达通知
    headerMD     metadata.MD   // 响应头元数据
    trailerMD    metadata.MD   // 响应尾元数据
    status       *status.Status // 响应状态
    buffer       *recvBuffer   // 接收缓冲区
    done         chan struct{}  // 流完成通知
}

序列化与反序列化

// gRPC-Go 的编解码（Codec）接口：
type Codec interface {
    Marshal(v interface{}) ([]byte, error)
    Unmarshal(data []byte, v interface{}) error
    Name() string
}

// 默认实现：protoCodec（Protobuf）
// 注册自定义编解码：
encoding.RegisterCodec(&JSONCodec{})

// 消息的序列化流程：
// 1. 客户端：Marshal(req) → 压缩（可选）→ 写入 HTTP/2 DATA 帧
// 2. 服务端：读取 HTTP/2 DATA 帧 → 解压（可选）→ Unmarshal(data, req)

// gRPC 消息帧格式（Length-Prefixed Message）：
// ┌──────────┬──────────────────────┐
// │ 1 byte   │ 4 bytes              │ N bytes
// │ Compressed│ Message Length       │ Message Data
// │ (0/1)    │ (大端序)             │ (Protobuf 序列化)
// └──────────┴──────────────────────┘

// 压缩标志：0=未压缩，1=已压缩
// 支持的压缩算法：gzip, deflate, snappy, zstd（需注册）

// 性能优化点：
// 1. 大消息使用压缩（gzip 可减少 70%+ 大小）
// 2. 小消息不压缩（压缩开销大于收益）
// 3. 使用 proto.Marshal 而非 json.Marshal（快 10x+）
// 4. 复用 proto.Message 对象（sync.Pool）

流式 RPC 实现原理

// 服务端流式 RPC 的实现：

// 服务端 Stream 接口
type ServerStream interface {
    SetHeader(metadata.MD) error
    SendHeader(metadata.MD) error
    SetTrailer(metadata.MD)
    SendMsg(m interface{}) error
    RecvMsg(m interface{}) error
    Context() context.Context
}

// 服务端流式处理流程：
// 1. 客户端发送 HEADERS 帧（创建流）+ DATA 帧（请求消息）
// 2. 服务端接收请求，调用 handler
// 3. handler 循环调用 stream.SendMsg() 发送多个响应
// 4. 每个响应是一个独立的 DATA 帧
// 5. handler 返回后，服务端发送 trailers（状态码）+ END_STREAM

// 客户端流式接收：
// client.Recv() 阻塞等待下一个消息
// 当服务端发送 trailers 后，Recv() 返回 io.EOF

// 背压（Backpressure）实现：
// - HTTP/2 流控窗口限制发送速率
// - 服务端 SendMsg() 在窗口满时会阻塞
// - 客户端 Recv() 消费数据后发送 WINDOW_UPDATE
// - 形成自然的背压机制

// 双向流式（Bidi Streaming）：
// - 客户端和服务端都可以随时 SendMsg/RecvMsg
// - 两个方向的流独立进行
// - 典型场景：实时聊天、远程终端、游戏同步

错误处理与状态码

// gRPC 错误模型：基于 HTTP/2 trailers + Status proto

// Status 结构（google.golang.org/genproto/googleapis/rpc/status）
type Status struct {
    Code    int32    // 状态码（0=OK, 1=Canceled, 2=Unknown...）
    Message string   // 错误消息
    Details []*anypb.Any  // 错误详情（可携带任意 proto 消息）
}

// 常用状态码：
// OK (0)：成功
// Canceled (1)：请求被取消
// Unknown (2)：未知错误
// InvalidArgument (3)：无效参数
// DeadlineExceeded (4)：超时
// NotFound (5)：资源不存在
// AlreadyExists (6)：资源已存在
// PermissionDenied (7)：权限不足
// ResourceExhausted (8)：资源耗尽（限流）
// FailedPrecondition (9)：前置条件失败
// Aborted (10)：操作中止
// OutOfRange (11)：超出范围
// Unimplemented (12)：未实现
// Internal (13)：内部错误
// Unavailable (14)：服务不可用（可重试）
// DataLoss (15)：数据丢失
// Unauthenticated (16)：未认证

// 创建带详情的错误：
func NotFoundError(id string) error {
    st := status.Newf(codes.NotFound, "user %s not found", id)
    // 添加错误详情
    detail := &errdetails.ResourceInfo{
        ResourceType: "user",
        ResourceName: id,
        Owner: "auth-service",
        Description: "user does not exist",
    }
    st, _ = st.WithDetails(detail)
    return st.Err()
}

// 客户端解析错误详情：
err := client.GetUser(ctx, req)
if err != nil {
    st := status.Convert(err)
    fmt.Printf("code=%d message=%s\n", st.Code(), st.Message())
    for _, detail := range st.Details() {
        switch d := detail.(type) {
        case *errdetails.ResourceInfo:
            fmt.Printf("resource: %s/%s\n", d.ResourceType, d.ResourceName)
        }
    }
}

// 可重试的错误码：
// - Unavailable (14)：服务暂时不可用，可重试
// - DeadlineExceeded (4)：超时，可重试（幂等操作）
// - ResourceExhausted (8)：限流，可重试（带退避）
// 不可重试的错误码：
// - InvalidArgument (3)：参数错误，重试也没用
// - NotFound (5)：不存在，重试也没用
// - PermissionDenied (7)：权限不足，重试也没用

gRPC-Go 常见陷阱

  • 陷阱1：每次请求创建新连接 → 全局复用 ClientConn，HTTP/2 多路复用
  • 陷阱2：忽略 context 超时 → 所有 RPC 调用都应该设置超时（context.WithTimeout）
  • 陷阱3：大消息导致 OOM → 设置 MaxRecvMsgSize，默认 4MB 可能不够
  • 陷阱4：流式 RPC 不关闭 → 客户端必须 Recv 到 io.EOF，否则连接泄漏
  • 陷阱5：错误码用错 → 用 status.Errorf 而非 fmt.Errorf，客户端才能正确解析状态码
  • 陷阱6：拦截器顺序错误 → ChainUnaryInterceptor 按顺序执行，日志应该在最外层
  • 陷阱7：protobuf 字段编号变更 → 字段编号一旦使用不能更改，否则兼容性破坏
