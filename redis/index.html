Redis

Redis（Remote Dictionary Server）是内存级键值存储，也是数据结构服务器，强项在缓存、消息队列、实时计数、会话存储等场景。2026 年 Redis 8.x 是主流版本，Redis Stack 7.x 提供搜索、JSON 等扩展能力。

1. 安装运行

# macOS 一键安装
brew install redis
brew services start redis
# 源码编译（Linux/Docker 宿主机）
wget https://download.redis.io/redis-stable.tar.gz
tar -xzvf redis-stable.tar.gz
cd redis-stable
make
# 编译成功后，可执行文件在 src/ 目录：
# redis-server     Redis 服务端
# redis-cli       命令行客户端
# redis-sentinel  哨兵模式
make install
# Docker 快速启动
docker run -d --name redis -p 6379:6379 redis:8.10

3. Redis-CLI 客户端

# 交互式终端（Tab 补全非常智能）
redis-cli

# 常用姿势
redis-cli -x set mypasswd < /etc/passwd           # 管道喂 stdin
redis-cli -r 100 -i 1 info | grep used_memory_human  # 循环执行
redis-cli --scan --pattern 'user:*'                # 全量扫描不阻塞
redis-cli --eval myscript.lua key1 key2 , arg1 arg2  # Lua 脚本
redis-cli -h 192.168.1.100 -p 6379 -a mypassword    # 远程连接
redis-cli -h localhost -p 6379 -n 1                 # 选择数据库 1
redis-cli PING                                       # 测试连接
redis-cli INFO                                       # 查看信息
redis-cli DBSIZE                                     # 查看键数量

小技巧：-x 管道喂数据、-r 重复执行（配 -i 间隔）、--scan 安全扫描——比 KEYS * 靠谱得多。

5. 数据类型（核心）

5.1 String（字符串）

# 基础操作
SET key value
GET key
MSET key1 value1 key2 value2
MGET key1 key2

# 数值操作
SET count 10
INCR count        # 11
DECR count        # 10
INCRBY count 5   # 15
DECRBY count 3   # 12

# 过期时间
SETEX key 60 value          # 60秒后过期
PSETEX key 60000 value      # 60000毫秒后过期
SET key value EX 60         # SET + EX 组合

# 原子操作
SETNX key value             # key 不存在才设置
SET key value XX            # key 存在才设置
GETSET key newvalue         # 获取旧值并设置新值

5.2 Hash（哈希）

# 哈希操作
HSET user:1 name "Alice" age "25"
HGET user:1 name
HMGET user:1 name age
HGETALL user:1

# 字段操作
HINCRBY user:1 age 1        # age + 1
HDEL user:1 age
HEXISTS user:1 name

# 批量操作
HMSET user:1 name "Alice" age "25" city "Beijing"

5.3 List（列表）

# 列表操作
LPUSH mylist "a" "b" "c"     # 左侧插入: [c,b,a]
RPUSH mylist "d" "e"         # 右侧插入: [c,b,a,d,e]

LPOP mylist                  # 左侧弹出: c
RPOP mylist                  # 右侧弹出: e

LRANGE mylist 0 -1           # 获取所有: [b,a,d]

# 阻塞操作
BLPOP mylist 0               # 阻塞弹出（消息队列）
BRPOP mylist 0

5.4 Set（集合）

# 集合操作
SADD tags "python" "redis" "golang"
SMEMBERS tags
SISMEMBER tags "python"      # 1 存在，0 不存在

# 集合运算
SADD set1 "a" "b" "c"
SADD set2 "b" "c" "d"
SDIFF set1 set2              # 差集: [a]
SINTER set1 set2             # 交集: [b,c]
SUNION set1 set2             # 并集: [a,b,c,d]

# 随机操作
SRANDMEMBER tags 2           # 随机获取2个
SPOP tags                    # 随机弹出

5.5 ZSet（有序集合）

# 有序集合
ZADD leaderboard 100 "Alice"
ZADD leaderboard 200 "Bob"
ZADD leaderboard 150 "Charlie"

ZRANGE leaderboard 0 -1 WITHSCORES
# 1) "Charlie" 150
# 2) "Alice" 100
# 3) "Bob" 200

ZREVRANGE leaderboard 0 -1 WITHSCORES  # 降序

# 排名
ZRANK leaderboard "Alice"      # 1 (0-based)
ZREVRANK leaderboard "Alice"   # 1 (倒序)

# 范围查询
ZRANGEBYSCORE leaderboard 100 150

5.6 Stream（流）消息队列

# Stream 操作（Redis 5.0+）
XADD mystream * field1 value1 field2 value2
XADD mystream * msg "hello"

XRANGE mystream - + COUNT 10
XREAD COUNT 2 STREAMS mystream 0

# 消费者组
XGROUP CREATE mystream mygroup 0
XREADGROUP GROUP mygroup consumer1 STREAMS mystream >
XACK mystream mygroup message-id

6. 持久化机制

# RDB 快照（默认）
save 900 1      # 900秒内1个key变化
save 300 10     # 300秒内10个key变化
save 60 10000   # 60秒内10000个key变化

# AOF 追加文件
appendonly yes
appendfsync everysec  # everysec 推荐，性能与安全平衡
# appendfsync always  # 每次写入都同步，最安全但最慢
# appendfsync no      # 操作系统决定

# 混合持久化（Redis 4.0+，8.x 默认）
aof-use-rdb-preamble yes

pipe.incr('counter')

异步客户端（Redis-py 4.2+）

import asyncio
import redis.asyncio as redis

async def main():
    r = redis.Redis(host='localhost', port=6379, decode_responses=True)
    
    # 异步操作
    await r.set('key', 'value')
    value = await r.get('key')
    
    # 管道
    async with r.pipeline() as pipe:
        await pipe.set('key1', 'value1')
        await pipe.get('key1')
        result = await pipe.execute()
    
    await r.close()

asyncio.run(main())

9.1 缓存

# 缓存模式
import redis
import json

r = redis.Redis(decode_responses=True)

def get_user(user_id):
    cache_key = f'user:{user_id}'
    
    # 先查缓存
    cached = r.get(cache_key)
    if cached:
        return json.loads(cached)
    
    # 查数据库
    user = db.query(f'SELECT * FROM users WHERE id = {user_id}')
    
    # 写入缓存，1小时过期
    if user:
        r.setex(cache_key, 3600, json.dumps(user))
    
    return user

# 缓存更新策略
def update_user(user_id, data):
    # 更新数据库

    r.delete(f'user:{user_id}')

9.2 分布式锁

# 分布式锁
import redis
import time

r = redis.Redis(decode_responses=True)

def acquire_lock(lock_name, timeout=10):
    lock_key = f'lock:{lock_name}'
    # SET NX EX 原子操作
    result = r.set(lock_key, '1', nx=True, ex=timeout)
    return result is not None

def release_lock(lock_name):
    lock_key = f'lock:{lock_name}'
    r.delete(lock_key)

# 使用
def process_order(order_id):
    lock_name = f'order:{order_id}'
    
    if acquire_lock(lock_name):
        try:
            # 处理订单
            pass
        finally:
            release_lock(lock_name)
    else:
        print('获取锁失败，订单正在处理中')

# UV 统计
def track_visit(page_id):
    key = f'uv:page:{page_id}:{date.today()}'

9.4 消息队列

# 生产者
def publish_task(task_data):
    r.lpush('task_queue', json.dumps(task_data))

# 消费者
def consume_task():
    while True:
        # 阻塞获取
        task = r.brpop('task_queue', timeout=0)
        if task:
            _, data = task
            process(json.loads(data))
