Go 中 gRPC 服务的工程实践

🏷️ Go gRPC

HTTP/2 的帧与流控、四种调用模式的边界、deadline 端到端传播的机制、错误模型里 status 与 details 的设计、keepalive 与中间代理造成的偶发断连，都在gRPC 的传输机制与工程落地里。这一篇不重复那些，只讲 Go 这个具体实现里那些会让你半夜起来的细节：版本、拦截器、停机、泄漏、测试。

代码生成：两个插件、两套版本

最常见的困惑是「装了 protoc-gen-go 却没生成 service 代码」。原因是 protobuf 的 Go 实现换到新 API 之后，protoc-gen-go 只负责消息类型，服务桩交给了独立的 protoc-gen-go-grpc，两个插件的版本号完全独立，必须分别安装。

组件负责什么版本与坑
protoc 或 buf解析 proto 并调用插件buf 能顺带做 lint 与破坏性变更检查，见总览篇
protoc-gen-go只生成消息类型新 API 之后不再生成任何 service 代码
protoc-gen-go-grpc生成 client 与 server 桩版本与上面那个无关，忘装的症状就是「找不到 RegisterXxxServer」
生成代码里的 SupportPackageIsVersionN编译期版本断言grpc-go 太旧会正好在这一行报错，该升 grpc-go 而不是改生成的文件
go_package 选项决定导入路径与输出目录写错会生成到意外的目录，或者两个 proto 撞包名
require_unimplemented_servers控制是否强制嵌入默认要求嵌入；改成不要求会把运行时问题变成编译时问题

嵌入 UnimplementedXxxServer 这件事值得单独想清楚，因为它是一个把编译期错误换成运行期错误的交易。

做法proto 新增方法时风险什么时候选它
嵌入（默认）编译通过，运行时返回 Unimplemented忘了实现不会有人发现，直到客户端报错proto 与服务端分开发布、必须向前兼容时
关掉强制嵌入编译失败一次 proto 变更会卡住所有实现方的构建单仓库、proto 和实现一起发版时
嵌入 + CI 断言编译通过，但 CI 报出未实现的方法要自己写这条检查推荐：既保留兼容性又不丢失信号
 
type orderServer struct {
	// 生成器默认要求嵌入它（靠一个 mustEmbed 方法强制）。代价要清楚：
	// 以后 proto 里加了方法、你忘了实现，照样编译通过，线上表现是客户端收到 Unimplemented。
	// 所以 proto 变更必须靠 CI 检查兜住，不能指望编译器报错。
	pb.UnimplementedOrderServiceServer
	repo *Repo
}

func main() {
	lis, err := net.Listen("tcp", ":9090")
	if err != nil {
		log.Fatalf("listen: %v", err)
	}

	s := grpc.NewServer(
		// 一元和流式是两条完全独立的链，recover 必须两边都挂，
		// 只挂一元链的话，流式接口里的 panic 会直接打挂进程。
		grpc.ChainUnaryInterceptor(recoveryUnary, logUnary),
		grpc.ChainStreamInterceptor(recoveryStream, logStream),

		// 接收上限默认 4 MiB，发送侧几乎不限。两端要对称调，
		// 否则一端发得出去、另一端收不下，客户端看到的是 ResourceExhausted。
		grpc.MaxRecvMsgSize(8 << 20),

		grpc.KeepaliveParams(keepalive.ServerParameters{
			// 让长连接定期主动过期，客户端重连时才会重新走一遍负载均衡。
			// 不设它，扩容出来的新实例可能长期一点流量都没有。
			MaxConnectionAge:      30 * time.Minute,
			MaxConnectionAgeGrace: 10 * time.Second,
			Time:                  2 * time.Hour,
			Timeout:               20 * time.Second,
		}),
		grpc.KeepaliveEnforcementPolicy(keepalive.EnforcementPolicy{
			// 客户端 ping 比这个更频繁会收到 GOAWAY 并被断开。默认是 5 分钟，
			// 客户端那边随手配个 30 秒 keepalive 就会被踢 —— 两边必须一起调。
			MinTime:             30 * time.Second,
			PermitWithoutStream: true,
		}),
	)

	pb.RegisterOrderServiceServer(s, &orderServer{repo: newRepo()})

	// 健康检查服务。K8s 1.27 起原生 gRPC 探针已 GA，可以直接探它，
	// 不必再往镜像里塞 grpc_health_probe 二进制。
	hs := health.NewServer()
	healthpb.RegisterHealthServer(s, hs)
	hs.SetServingStatus("", healthpb.HealthCheckResponse_SERVING)

	go func() {
		if err := s.Serve(lis); err != nil {
			log.Fatalf("serve: %v", err)
		}
	}()

	stop := make(chan os.Signal, 1)
	signal.Notify(stop, syscall.SIGTERM, syscall.SIGINT)
	<-stop

	// 第一步：先置为 NOT_SERVING，让探针和注册中心开始摘流量。
	// 这一步必须早于 GracefulStop，否则摘流窗口期内打进来的新请求会直接被拒。
	hs.SetServingStatus("", healthpb.HealthCheckResponse_NOT_SERVING)
	time.Sleep(5 * time.Second) // 等端点更新传播完，时长按集群实测定，不是拍的

	// 第二步：GracefulStop 停止接受新连接、发 GOAWAY、等在途调用结束。
	// 它会一直等 —— 有长连流就等于永不退出，必须带超时兜底。
	done := make(chan struct{})
	go func() {
		s.GracefulStop()
		close(done)
	}()
	select {
	case <-done:
	case <-time.After(20 * time.Second):
		log.Println("graceful stop timeout, force stop")
		s.Stop() // 强制取消所有在途调用，客户端会收到 Unavailable
	}
}

一次调用在 Go 侧穿过哪些层

把这张图记住，后面所有「这段逻辑该写在哪」的问题都能直接回答。

flowchart TB APP["业务代码调用生成出来的 client 方法"] --> CLI subgraph CLI["客户端侧"] C1["客户端拦截器链：一元与流式各一条"] C2["service config：超时、重试、负载均衡策略"] C3["名称解析与 picker 选出一个子连接"] C4["StatsHandler：链路追踪在这里注入"] end CLI --> NET["HTTP/2 连接。帧与流控细节见 gRPC 总览篇"] NET --> SRV subgraph SRV["服务端侧"] S1["transport 收帧并为这次调用新建一个 stream"] S2["StatsHandler：解出链路上下文"] S3["拦截器链：recover、鉴权、限流、日志"] S4["handler：拿到 ctx 与请求对象"] end SRV --> BIZ["业务逻辑。ctx 必须继续往下传，否则取消到这里就断了"]

两个位置关系要记牢：StatsHandler 比拦截器链更靠外，所以链路上下文在你的日志拦截器运行之前就已经解出来了；而service config 里的重试发生在客户端拦截器之内，你在客户端拦截器里看到的是一次逻辑调用，看不到它底下重试了几次。

context：取消怎么才算真的传到底

服务端 handler 拿到的 ctx 会在两种情况下被取消：客户端主动断开，或者 deadline 到期。但取消只有被一路传下去才有意义。把 ctx 换成 context.Background() 去查数据库，是 Go 服务里最普遍的一类浪费——客户端早就超时返回了，服务端还在慢查询上占着连接和内存。

反方向也有个坑：handler 一返回，它的 ctx 立刻被取消。所以在 handler 里 go 出去一个异步任务、还把这个 ctx 带上，那个任务几乎必然立刻失败。要做异步收尾，用 context.WithoutCancel 剥掉取消信号（Go 1.21 起），或者干脆投给一个有自己生命周期的后台队列。至于流式方法，stream.Context() 的生命周期和这次调用绑定，同样不能带出去。

拦截器：两条链、顺序与 recover

grpc-go 不会替你兜 panic：handler 在服务流的 goroutine 里跑，panic 一路冒上去就是进程退出。这一点和多数 HTTP 框架的默认行为相反，是 Go gRPC 服务上线初期最常见的事故来源。更容易漏的是一元和流式是两条完全独立的链，只在一元链上挂了 recover，流式方法照样能把进程打挂。

类型能拿到什么适合做做不到
一元服务端拦截器ctx、完整请求对象、方法名、handler鉴权、限流、参数审计、recover、改响应对流式方法完全无效
流式服务端拦截器包装用的 ServerStream、方法名包装流以逐条拦消息、recover、限流拿不到完整请求体，消息是一条条来的
一元客户端拦截器方法名、请求与响应指针、连接注入 metadata、熔断、本地降级看不见底层重试了几次
流式客户端拦截器流描述、streamer包装 ClientStream 做埋点同上
StatsHandler逐条消息的生命周期事件指标与链路追踪，流式观测唯一准确的方式不能改请求，也不能中断调用

顺序规则：链式选项里先传入的在最外层，所以 recover 应该放第一个，让它能兜住后面所有拦截器和 handler 的 panic。还有一条边界要写进团队规范：recover 只救得了同一个 goroutine，handler 里自己 go 出去的协程发生 panic 依然会打挂进程，这类地方必须各自带 recover。

// recover 要分别挂在两条链上。注意它只救得了同一个 goroutine：
// handler 里 go 出去的协程发生 panic，照样打挂整个进程。
func recoveryUnary(ctx context.Context, req any, info *grpc.UnaryServerInfo,
	handler grpc.UnaryHandler) (resp any, err error) {
	defer func() {
		if r := recover(); r != nil {
			// 堆栈只进日志。绝不要把 r 的内容拼进返回给客户端的 message。
			log.Printf("panic in %s: %v\n%s", info.FullMethod, r, debug.Stack())
			err = status.Error(codes.Internal, "internal error")
		}
	}()
	return handler(ctx, req)
}

func recoveryStream(srv any, ss grpc.ServerStream, info *grpc.StreamServerInfo,
	handler grpc.StreamHandler) (err error) {
	defer func() {
		if r := recover(); r != nil {
			log.Printf("panic in %s: %v\n%s", info.FullMethod, r, debug.Stack())
			err = status.Error(codes.Internal, "internal error")
		}
	}()
	return handler(srv, ss)
}

func logUnary(ctx context.Context, req any, info *grpc.UnaryServerInfo,
	handler grpc.UnaryHandler) (any, error) {
	start := time.Now()
	resp, err := handler(ctx, req)
	code := status.Code(err) // err 为 nil 时返回 codes.OK，不用自己判空

	// 关键：客户端主动取消和客户端超时都不是服务端故障。
	// 把它们混进错误率里，告警会长期误报，真故障反而被淹没。
	switch code {
	case codes.OK:
	case codes.Canceled, codes.DeadlineExceeded:
		metrics.ClientAbort(info.FullMethod, code.String())
	default:
		metrics.ServerError(info.FullMethod, code.String())
	}
	metrics.Latency(info.FullMethod, code.String(), time.Since(start))
	return resp, err
}

func (s *orderServer) GetOrder(ctx context.Context, req *pb.GetOrderRequest) (*pb.Order, error) {
	// 只有把这个 ctx 一路传到数据库和下游 HTTP，取消和 deadline 才会真的生效。
	// 中途换成 context.Background() 取数，等于把上游的超时全丢了，
	// 客户端早就超时返回，服务端还在慢查询上耗着连接。
	o, err := s.repo.Find(ctx, req.GetId())
	switch {
	case errors.Is(err, sql.ErrNoRows):
		// 必须转成 status。直接 return err 的话客户端看到的是 Unknown，
		// 而且 err.Error() 原样发给客户端，表名和 SQL 片段都可能泄露。
		return nil, status.Errorf(codes.NotFound, "order %s not found", req.GetId())
	case errors.Is(err, context.Canceled):
		return nil, status.Error(codes.Canceled, "canceled by client")
	case err != nil:
		log.Printf("find order %s: %v", req.GetId(), err) // 细节只进日志
		return nil, status.Error(codes.Internal, "internal error")
	}

	// 反例：handler 返回后 ctx 立刻被取消，别把它交给异步任务。
	// 确实要做收尾，用 context.WithoutCancel(ctx) 或者投给独立的后台队列。
	return o, nil
}

错误：别把 error 直接返回给客户端

从 handler 里 return err 有两个后果：客户端看到的码是 Unknown，没法做任何分支也没法配重试；并且 err.Error() 会原样发给客户端，数据库报错里的表名、SQL 片段、内部地址全都可能泄露。转换必须在边界上统一做。

codeGo 侧的典型来源算服务端故障吗客户端该做什么
Canceled客户端断开，或上游 ctx 被取消不算不重试、不告警，指标要单独分类
DeadlineExceeded客户端设的 deadline 到期要看是谁慢只有幂等方法才可以重试
InvalidArgument参数校验失败不算改代码，重试无意义
NotFound业务查不到不算走业务分支
ResourceExhausted限流命中，或消息超过接收上限两种都有退避重试；但消息体超限重试永远不会成功
Unavailable连接断开、收到 GOAWAY、服务正在停机算最该配自动重试的一档
Unimplemented忘了实现、方法名不匹配、代理路由错算查版本与路由，重试没用
Internalpanic 被 recover、序列化失败算不要自动重试
Unknownhandler 直接 return 了普通 error算这是服务端的 bug，去补 status 转换

客户端侧的判读有个细节常被忽略：status.FromError 的第二个返回值表示「这个 error 到底是不是服务端返回的 status」。为 false 时说明它是本地错误（连不上、序列化失败、本地 ctx 取消），此时拿到的码只是被填成了 Unknown 或 Canceled，不能用来做业务分支。另外，把 status 用 fmt.Errorf 包一层之后还能不能被识别出来，取决于 grpc-go 版本——不要依赖这个解包行为，跨层传递时要么原样返回，要么显式重建 status。

客户端：惰性连接与 service config

grpc.Dial 已经废弃，换成 grpc.NewClient。这次替换不是改个名字，有两个行为差异会直接让代码跑不起来。

第一，它不建连就返回，通道处于 IDLE，第一次 RPC 才触发连接。因此 err == nil 完全不代表服务端活着，拿它当启动期健康检查是错的；想预热就显式调 Connect()，因为 NewClient 不支持 WithBlock 那一组选项。第二，默认解析器从 passthrough 变成了 dns，目标串会走 DNS 解析路径；测试里用自定义拨号器时必须显式写 passthrough:/// 前缀，否则它会真的去解析一个不存在的域名——这是从 Dial 迁过来时最典型的翻车点。

stateDiagram-v2 state "IDLE：NewClient 返回后就在这里，还没建连" as Idle state "CONNECTING：正在解析与握手" as Conn state "READY：可以发起调用" as Ready state "TRANSIENT_FAILURE：失败，按退避等待重连" as TF state "SHUTDOWN：Close 之后" as Down [*] --> Idle Idle --> Conn: 第一次调用，或显式 Connect Conn --> Ready: 握手成功 Conn --> TF: 解析失败或连接被拒 TF --> Conn: 退避结束自动重连 Ready --> TF: 连接断开 Ready --> Idle: 空闲超时，或收到 GOAWAY Idle --> Down: Close Ready --> Down: Close TF --> Down: Close Down --> [*]

停在 TRANSIENT_FAILURE 时，默认行为是快速失败：调用立刻返回 Unavailable 而不是等着重连。加 grpc.WaitForReady(true) 会改成阻塞到通道就绪或 deadline 到期——它把「连不上」从立即失败变成了超时，用在启动期依赖上合适，用在在线请求上会放大故障时的排队。

// service config 随连接下发，按方法配。重试绝不能一刀切全开。
const svcCfg = `{
  "methodConfig": [{
    "name": [{"service": "order.v1.OrderService", "method": "GetOrder"}],
    "timeout": "2s",
    "retryPolicy": {
      "maxAttempts": 4,
      "initialBackoff": "0.05s",
      "maxBackoff": "1s",
      "backoffMultiplier": 2,
      "retryableStatusCodes": ["UNAVAILABLE"]
    }
  }],
  "retryThrottling": {"maxTokens": 100, "tokenRatio": 0.1},
  "loadBalancingConfig": [{"round_robin": {}}]
}`

func newClient(target string) (pb.OrderServiceClient, func() error, error) {
	// NewClient 取代了已废弃的 Dial。两个必须知道的行为差异：
	// 1) 它不建连就返回，通道处于 IDLE，第一次调用才触发连接。
	//    所以这里 err 为 nil 不代表服务端活着，别拿它当健康检查。
	// 2) 默认解析器是 dns（Dial 是 passthrough），写 "127.0.0.1:9090" 也会走 DNS 路径；
	//    要配自定义拨号器必须显式写 "passthrough:///xxx"。
	cc, err := grpc.NewClient(target,
		grpc.WithTransportCredentials(insecure.NewCredentials()),
		grpc.WithDefaultServiceConfig(svcCfg),
		grpc.WithStatsHandler(otelgrpc.NewClientHandler()),
		grpc.WithKeepaliveParams(keepalive.ClientParameters{
			Time:                30 * time.Second, // 不能小于服务端的 MinTime，否则被 GOAWAY 踢掉
			Timeout:             10 * time.Second,
			PermitWithoutStream: true,
		}),
	)
	if err != nil {
		return nil, nil, err
	}
	cc.Connect() // 想启动时预热就显式调它；NewClient 不支持 WithBlock
	return pb.NewOrderServiceClient(cc), cc.Close, nil
}

func getOrder(ctx context.Context, c pb.OrderServiceClient, id string) (*pb.Order, error) {
	// 客户端默认没有 deadline。不设的话，服务端一卡这个调用就一直挂着。
	ctx, cancel := context.WithTimeout(ctx, 2*time.Second)
	defer cancel()

	o, err := c.GetOrder(ctx, &pb.GetOrderRequest{Id: id})
	if err != nil {
		// 第二个返回值的含义是「这个 error 到底是不是服务端返回的 status」。
		// 为 false 说明是本地错误（连不上、序列化失败、本地 ctx 取消），
		// 这时候拿 code 去做业务分支是错的 —— 它只是被填成了 Unknown 或 Canceled。
		st, ok := status.FromError(err)
		if !ok {
			return nil, fmt.Errorf("local failure: %w", err)
		}
		// 另外：用 fmt.Errorf 把 status 包一层之后还能不能被识别，取决于 grpc-go 版本。
		// 跨层传递时不要依赖解包，要么原样返回，要么显式重建一个 status。
		switch st.Code() {
		case codes.NotFound:
			return nil, ErrOrderNotFound
		case codes.Unavailable, codes.DeadlineExceeded:
			// 走到这里说明 service config 的重试已经用完了，这是最终失败
			return nil, ErrRetryLater
		}
		return nil, fmt.Errorf("%s: %s", st.Code(), st.Message())
	}
	return o, nil
}
service config 字段含义容易配错的地方
maxAttempts总尝试次数，包含第一次它不是「重试次数」；实现上还有个硬上限，配再大也无效
三个 backoff 字段指数退避退避总时长受调用 deadline 约束，deadline 一到就不再试
retryableStatusCodes哪些码触发重试别把 Internal 放进去；DeadlineExceeded 也几乎没有意义
retryThrottling整体失败率高时全局压制重试不配的话，下游一抖动重试就把流量放大数倍
hedgingPolicy并行发多份取最先返回的与 retryPolicy 互斥，而且只能用于幂等方法
重试的生效窗口只在还没收到服务端响应头之前所以流式调用里，自动重试基本只在建流阶段有用

流式接口的资源释放与泄漏排查

流式调用的泄漏几乎都来自同一个误解：以为流会自己关。客户端侧只要没读到结束标志就提前退出循环，这条流在传输层依然存在，读循环协程也不会退出。标准写法是创建流之前用可取消的上下文包一层并立刻 defer cancel()，这样无论从哪个分支返回都会把流关掉。服务端侧对应的规则是：handler 一返回流就结束，所以在 handler 里起的协程必须在返回前收敛，否则它们会拿着一个已取消的上下文继续跑。

现象查什么怎么判读
goroutine 数单调上涨pprof 的 goroutine 栈（debug=2）大量栈停在流的接收调用上，是客户端流没关；停在自己的 channel 上，是 handler 漏了
内存随运行时间线性涨goroutine + heap 两个一起看每条未关闭的流都占着收发缓冲，量级和流数成正比
连接数正常但活跃流很多channelz单个连接上的活跃流数只增不减，就是泄漏
客户端偶发 Unavailable提高 grpc-go 自身日志级别的两个环境变量看是不是 ping 过于频繁被 GOAWAY，或者服务端在滚动重启
没有请求但 CPU 不降pprof 的 CPU profile常见是 handler 返回后仍在跑的后台循环

优雅停机：GracefulStop 与 K8s 的配合

GracefulStop 的语义是：关闭监听、给已有连接发 GOAWAY、然后一直等到所有在途调用结束。问题就在「一直等」上——长生命周期的流式调用不会自己结束，于是进程永远不退，最后被编排系统强杀。

sequenceDiagram participant K as kubelet participant P as preStop 与信号处理 participant S as grpc.Server participant C as 客户端 K->>P: 发 SIGTERM，同时开始删 Endpoints P->>S: 先把健康状态置成 NOT_SERVING Note over K,C: 端点删除是异步的
这段窗口里仍然会有新请求打进来 P->>P: 等一段固定时长，让摘流传播完 P->>S: 调用 GracefulStop S->>C: 发 GOAWAY，不再接受新流 C->>C: 重新解析并连到别的实例 S->>S: 等在途调用自然结束 Note over S,C: 长连流不会自己结束
必须由业务收到停机信号后主动关 P->>S: 超时仍未退出，调用 Stop 强制结束 S-->>C: 在途调用收到 Unavailable
环节参数或动作默认调错的后果
宽限期terminationGracePeriodSeconds30 秒小于在途请求耗时就会被强杀，请求被截断且日志来不及落
摘流窗口preStop 睡一会，或主动置 NOT_SERVING无不留窗口则端点还没更新完就已拒绝新请求，客户端看到 Unavailable
就绪探针原生 gRPC 探针（K8s 1.27 起 GA）无用 exec 方式塞探测二进制会增加镜像体积和每次探测的开销
GracefulStop没有超时参数，会一直等—有长连流就永不退出，必须自己套超时
Stop立即取消所有在途调用—直接调它等于让在途请求全部失败
MaxConnectionAge服务端 keepalive 参数不限不设则老连接永不重连，扩容出来的实例长期没有流量

还有一个容易被忽略的前提：gRPC 是长连接，摘掉 Endpoints 并不会断开已建立的连接。真正让客户端换实例的是 GOAWAY。所以顺序必须是「先让新连接不再选到我（摘流），再对老连接发 GOAWAY（GracefulStop）」，反过来做就会在窗口期里制造一批失败请求。

测试与可观测的接入位置

测试用 bufconn：在进程内跑真实的 gRPC 栈，真编解码、真拦截器链、真 status 码，不占端口也不会有端口冲突。它比手写一个 mock client 有价值得多，因为拦截器链也被覆盖到了——鉴权拦截器把错误吞成 OK 这类问题，只有走真实栈才测得出来。

可观测这边，用 StatsHandler 而不是拦截器接 OpenTelemetry：官方的拦截器式接法已标记废弃，而 StatsHandler 能拿到逐条消息的事件，流式调用的观测才准确；它又比拦截器链更靠外，所以你在日志拦截器里可以直接取到已经解好的追踪标识。代价是它只能观测、不能修改请求或中断调用，那些仍然要放在拦截器里做。

// bufconn：在进程内跑真实的 gRPC 栈 —— 真编解码、真拦截器链、真 status 码，
// 不占端口也不会有端口冲突。比手写 mock client 有价值得多，因为它能测到拦截器。
func startTestServer(t *testing.T, srv pb.OrderServiceServer) pb.OrderServiceClient {
	t.Helper()
	lis := bufconn.Listen(1024 * 1024)
	s := grpc.NewServer(
		grpc.ChainUnaryInterceptor(recoveryUnary, logUnary), // 拦截器要一起测，否则线上才发现它吞了错误
		grpc.StatsHandler(otelgrpc.NewServerHandler()),
	)
	pb.RegisterOrderServiceServer(s, srv)
	go func() { _ = s.Serve(lis) }()

	// 关键的一行：NewClient 默认走 dns 解析器，必须写 passthrough 前缀，
	// 否则它会真的去解析 "bufnet" 这个域名然后失败。用旧的 Dial 时没这个问题。
	cc, err := grpc.NewClient("passthrough:///bufnet",
		grpc.WithContextDialer(func(ctx context.Context, _ string) (net.Conn, error) {
			return lis.DialContext(ctx)
		}),
		grpc.WithTransportCredentials(insecure.NewCredentials()),
	)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = cc.Close(); s.Stop(); _ = lis.Close() })
	return pb.NewOrderServiceClient(cc)
}

// 服务端流的客户端侧：不读到结束就退出会泄漏，必须靠 cancel 收尾。
func firstN(ctx context.Context, c pb.OrderServiceClient, n int) ([]*pb.Order, error) {
	// 这两行是流式调用的标准写法。少了 defer cancel()，一旦提前 break，
	// 这条流在传输层不会关闭，读循环协程也不退出，表现就是协程数单调上涨。
	ctx, cancel := context.WithCancel(ctx)
	defer cancel()

	stream, err := c.ListOrders(ctx, &pb.ListOrdersRequest{})
	if err != nil {
		return nil, err
	}
	var out []*pb.Order
	for len(out) < n {
		o, err := stream.Recv()
		if errors.Is(err, io.EOF) { // 服务端正常结束了流
			break
		}
		if err != nil {
			return nil, err
		}
		out = append(out, o)
	}
	return out, nil // 提前返回是安全的，因为 defer cancel() 会把流关掉
}
❌
最容易踩的坑
  • 只在一元链上挂了 recover。流式方法里的 panic 会直接打挂进程，两条链是独立的。
  • 以为 grpc-go 会帮你兜 panic。它不会；而且 recover 只救同一个 goroutine，handler 里 go 出去的协程要各自带。
  • handler 里 return err。客户端只看到 Unknown，并且内部错误文字会原样发出去。
  • 把 Canceled 和 DeadlineExceeded 算进服务端错误率。那是客户端行为，会让告警长期误报。
  • 业务层用 context.Background() 取数。上游的取消与 deadline 到这里就断了，客户端早返回了服务端还在跑。
  • 把 handler 的 ctx 带进异步任务。handler 一返回它就被取消；要剥离取消信号或改用独立的后台队列。
  • 迁到 NewClient 后仍用自定义拨号器却不写 passthrough:///。默认解析器变成了 dns，它会去解析一个不存在的域名。
  • 拿 NewClient 不报错当成服务端存活。它是惰性的，连接要到第一次调用才建立。
  • 客户端不设 deadline。grpc-go 默认没有超时，服务端一卡调用就一直挂着。
  • 只在服务端配了 keepalive 强制策略，客户端 ping 更频繁。会被 GOAWAY 断开，表现为周期性偶发失败。
  • 流式调用没有 defer cancel() 就提前 break。流不会关，协程和缓冲一起泄漏。
  • 把 maxAttempts 当成重试次数。它包含第一次调用，而且实现上还有硬上限。
  • 配了重试却不配 retryThrottling。下游一抖动，重试会把流量放大好几倍。
  • 直接调 Stop，或者 GracefulStop 不带超时。前者截断在途请求，后者遇到长连流就永不退出。
  • 停机时先 GracefulStop 再摘流。顺序反了，窗口期内会有一批请求确定失败。
  • 用 mock 出来的 client 做测试。拦截器、编解码、status 映射全都测不到，用 bufconn 跑真实栈。
