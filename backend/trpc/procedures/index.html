Procedures 与路由

三种 Procedure

类型用途HTTP 方法
Query获取数据GET
Mutation修改数据POST/PUT/DELETE
Subscription实时推送WebSockets
 
export const appRouter = t.router({
  // 查询
  getUsers: t.procedure.query(() => {
    return db.user.findMany()
  }),
  
  // 带输入的查询
  getUserById: t.procedure
    .input(z.string())
    .query(({ input }) => {
      return db.user.findUnique({ where: { id: input } })
    }),
  
  // 修改
  createUser: t.procedure
    .input(z.object({ name: z.string(), email: z.string().email() }))
    .mutation(({ input }) => {
      return db.user.create({ data: input })
    }),
})

路由组织

// routers/users.ts
const usersRouter = t.router({
  list: t.procedure.query(() => ...),
  getById: t.procedure.input(z.string()).query(() => ...),
  create: t.procedure.input(CreateUserSchema).mutation(() => ...),
})

// routers/posts.ts
const postsRouter = t.router({
  list: t.procedure.query(() => ...),
  create: t.procedure.input(CreatePostSchema).mutation(() => ...),
})

// 主路由
export const appRouter = t.router({
  users: usersRouter,
  posts: postsRouter,
})

// 前端调用
client.users.list.query()
client.posts.create.mutation({ title: 'Hello' })

中间件

const isAuthed = t.middleware(({ ctx, next }) => {
  if (!ctx.user) throw new TRPCError({ code: 'UNAUTHORIZED' })
  return next({ ctx: { user: ctx.user } })
})

const protectedProcedure = t.procedure.use(isAuthed)

const postsRouter = t.router({
  create: protectedProcedure
    .input(CreatePostSchema)
    .mutation(({ ctx, input }) => {
      // ctx.user 一定存在
      return db.post.create({ data: { ...input, authorId: ctx.user.id } })
    }),
})
