柔性事务的实现细节与失败处理

🏷️ 分布式事务

2PC 的不确定态、TCC 与 Saga 与本地消息表与事务消息的四方案对照、以及选型判断，分布式系统主干篇已经给出，本篇不再重复对比。这里只讲一件事：选定方案之后，表结构怎么设计、代码怎么写、失败路径怎么收尾。柔性事务上线后出的事故，几乎没有一个是因为选错了方案，全部来自这些细节——消息表没加唯一键、投递任务没做抢占、补偿接口不幂等、没有巡检、没有对账。

本地消息表：DDL 决定了这个方案能不能用

本地消息表的可靠性完全建立在表结构上。缺了 biz_id 唯一键就防不住重复插入，缺了 next_retry_at 就只能固定间隔重试，缺了扫描索引就会在数据量涨起来后把业务库扫崩。

CREATE TABLE `tx_message` (
  `id`            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `biz_id`        VARCHAR(64)  NOT NULL COMMENT '业务唯一标识，如 order_paid:20260923001',
  `topic`         VARCHAR(64)  NOT NULL,
  `tag`           VARCHAR(64)  NOT NULL DEFAULT '',
  `payload`       VARCHAR(2048) NOT NULL COMMENT '超过 2KB 的放对象存储，只存引用',
  `status`        TINYINT      NOT NULL DEFAULT 0 COMMENT '0 待投递 1 投递中 2 已投递 3 死信',
  `retry_count`   INT          NOT NULL DEFAULT 0,
  `next_retry_at` DATETIME(3)  NOT NULL COMMENT '下次可投递时间，插入时即为当前时间',
  `owner`         VARCHAR(64)  NOT NULL DEFAULT '' COMMENT '抢占者实例标识，用于排查与超时回收',
  `locked_at`     DATETIME(3)  NULL COMMENT '抢占时间，超时回收依据',
  `created_at`    DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  `updated_at`    DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3),
  PRIMARY KEY (`id`),
  -- 这条唯一键是整个方案的地基：业务重试时重复插入会直接冲突，天然幂等
  UNIQUE KEY `uk_biz_id` (`biz_id`),
  -- 扫描索引必须把 status 放最左，next_retry_at 次之，避免扫到已投递的历史数据
  KEY `idx_scan` (`status`, `next_retry_at`),
  -- 归档与超时回收用
  KEY `idx_created` (`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='本地消息表';
字段省掉会怎样取值建议
biz_id 唯一键业务重试时插入两条，下游收到两次且无法区分业务类型加业务主键拼成，禁止用自增或随机串
status 含「投递中」多实例无法抢占，必然重复投递四态足够：待投递、投递中、已投递、死信
next_retry_at只能固定间隔重试，下游故障时被持续打压退避序列 1s、5s、30s、2m、10m、30m、1h
retry_count无法判断何时放弃，坏消息永久重试上限 8 到 12 次，超限转死信并告警
owner 与 locked_at抢占后进程崩溃，记录永久卡在投递中回收阈值取单条投递超时的 3 倍，典型 2 到 5 分钟
idx_scan 联合索引扫描退化成全表扫，消息表涨到千万级后压垮业务库status 最左，配合分批 limit

投递任务：抢占、退避、死信，三件都不能少

业务与消息必须在同一个本地事务里写入，这是方案的前提；而投递任务的正确性关键在抢占。下面这段是可以直接抄的骨架：

@Service
public class OrderService {

    // 关键点：业务表与消息表在同一个本地事务内，靠数据库保证两者同生共死。
    // 绝对不要在这个事务里直接调 MQ 发送——网络调用会拉长事务、占住行锁，
    // 而且发送成功但事务回滚会造成下游收到不存在的业务。
    @Transactional(rollbackFor = Exception.class)
    public void payOrder(long orderId, long amount) {
        int rows = orderMapper.markPaid(orderId); // UPDATE ... WHERE status = 'UNPAID'
        if (rows == 0) {
            throw new BizException("订单状态不允许支付，可能已被处理");
        }
        TxMessage msg = new TxMessage();
        msg.setBizId("order_paid:" + orderId);   // 与 uk_biz_id 对应，重复调用直接冲突
        msg.setTopic("ORDER_PAID");
        msg.setPayload(JSON.toJSONString(Map.of("orderId", orderId, "amount", amount)));
        msg.setNextRetryAt(new Date());          // 立即可投
        try {
            txMessageMapper.insert(msg);
        } catch (DuplicateKeyException e) {
            // 同一业务重复进来：消息已存在，无需再插，整体保持幂等
            log.info("tx message exists, bizId={}", msg.getBizId());
        }
    }
}

@Component
public class TxMessageDispatcher {

    private static final int[] BACKOFF_SEC = {1, 5, 30, 120, 600, 1800, 3600, 7200};
    private final String owner = InstanceId.get(); // 形如 hostname-pid

    @Scheduled(fixedDelay = 1000)
    public void dispatch() {
        // 1) 分批扫描，limit 控制单次影响面，避免长事务与大结果集
        List<TxMessage> batch = mapper.scanPending(200);
        for (TxMessage m : batch) {
            // 2) 乐观锁抢占：只有把 status 从 0 改到 1 成功的实例才发送。
            //    多实例同时扫到同一条，只有一个 UPDATE 能命中，其余受影响行数为 0。
            if (mapper.acquire(m.getId(), owner) == 0) {
                continue;
            }
            try {
                producer.send(m.getTopic(), m.getTag(), m.getPayload());
                mapper.markSent(m.getId());
            } catch (Exception e) {
                int next = m.getRetryCount() + 1;
                if (next >= BACKOFF_SEC.length) {
                    // 3) 超过上限转死信并告警，绝不静默丢弃
                    mapper.markDead(m.getId());
                    alarm.fire("tx_message_dead", m.getBizId(), e.getMessage());
                } else {
                    // 4) 指数退避，把下次投递时间推后，避免持续打压故障下游
                    mapper.releaseWithBackoff(m.getId(), next, BACKOFF_SEC[next]);
                }
            }
        }
    }

    // 5) 超时回收：抢占后进程崩溃的记录会卡在 status=1，必须捞回来
    @Scheduled(fixedDelay = 60000)
    public void reclaim() {
        int n = mapper.reclaimStale(180); // locked_at 早于 180 秒前的改回 status=0
        if (n > 0) {
            log.warn("reclaimed {} stale tx messages", n);
        }
    }
}

SELECT * FROM tx_message
 WHERE status = 0 AND next_retry_at <= NOW(3)
 ORDER BY next_retry_at
 LIMIT 200;

-- 乐观锁抢占：受影响行数为 1 才算抢到，这一条是防重复投递的核心
UPDATE tx_message
   SET status = 1, owner = 'host-1234', locked_at = NOW(3)
 WHERE id = 10086 AND status = 0;

-- 退避释放：把下次可投时间往后推
UPDATE tx_message
   SET status = 0, retry_count = 3,
       next_retry_at = DATE_ADD(NOW(3), INTERVAL 120 SECOND),
       owner = '', locked_at = NULL
 WHERE id = 10086 AND status = 1;

-- 超时回收：抢占者崩溃留下的孤儿记录，LIMIT 控制单次删改规模
UPDATE tx_message
   SET status = 0, owner = '', locked_at = NULL
 WHERE status = 1 AND locked_at < DATE_SUB(NOW(3), INTERVAL 180 SECOND)
 LIMIT 500;

消息表一定会膨胀，日均百万单的系统一年就是几亿行。归档策略要在上线前定好，等表已经几千万行再动就只能停机处理：

归档方案做法优势代价
按天 RANGE 分区 + DROP PARTITION按 created_at 分区，保留 30 天，过期分区直接 drop清理是元数据操作，秒级完成，不产生 undo 与大量 binlog需要提前建分区的运维任务，漏建会导致写入失败
定时搬到归档表每晚把已投递且超保留期的记录批量搬走再删不依赖分区特性，兼容所有版本大批量删除产生大量 binlog，必须小批加睡眠否则从库延迟飙升
直接分批 DELETE按时间范围每批几百行删实现最简单InnoDB 空间不回收，需额外做表重建
冷数据导出对象存储归档前导出到对象存储，库内只留近期成本最低，满足审计留存查历史要走离线流程
⚠️
死信不是终点

转死信只解决了「不再无谓重试」，没解决业务不一致。死信必须触发告警并进入人工处理流程，告警内容要带 biz_id 与失败原因摘要。见过太多系统死信表堆了几万条没人看，等到对账才发现下游少了一个月的数据。

事务消息：回查接口的前提是本地事务留痕

事务消息（RocketMQ 的半消息机制）看起来省掉了消息表，但它把可靠性押在回查接口上。而回查能不能答得出来，取决于本地事务是否留下了可按事务 ID 查询的痕迹——这是最常被漏掉的前提。

sequenceDiagram autonumber participant P as 生产者 participant B as Broker participant D as 业务库 participant C as 消费者 P->>B: 发送半消息，对消费者不可见 B-->>P: 半消息已存储 P->>D: 执行本地事务，同时写入带事务 ID 的留痕记录 alt 本地事务提交成功 P->>B: 二次确认：提交 B->>C: 消息正式投递 else 二次确认丢失，进程崩溃或网络断开 Note over B: 半消息超时未确认，Broker 主动发起回查 B->>P: 回查：该事务 ID 的最终状态 P->>D: 按事务 ID 查留痕记录 alt 查到记录 P-->>B: 回答提交 B->>C: 消息正式投递 else 查不到记录 P-->>B: 回答回滚，半消息丢弃 end end Note over P,D: 若本地事务没有留痕，只能回答未知
半消息被反复回查直至超限丢弃，下游静默少数据 
public class PayTransactionListener implements TransactionListener {

    @Override
    public LocalTransactionState executeLocalTransaction(Message msg, Object arg) {
        String txId = msg.getTransactionId();
        try {
            // 关键：本地事务内必须写一条以 txId 为唯一键的留痕记录，
            // 否则回查时无从判断。这条记录是本地事务是否提交的唯一凭据。
            orderService.payOrderWithTrace((PayCmd) arg, txId);
            return LocalTransactionState.COMMIT_MESSAGE;
        } catch (DuplicateKeyException e) {
            // 同一 txId 已处理过：视为已提交，保持幂等
            return LocalTransactionState.COMMIT_MESSAGE;
        } catch (Exception e) {
            log.error("local tx failed, txId={}", txId, e);
            return LocalTransactionState.ROLLBACK_MESSAGE;
        }
    }

    @Override
    public LocalTransactionState checkLocalTransaction(MessageExt msg) {
        String txId = msg.getTransactionId();
        // 回查必须只读且幂等：Broker 默认最多回查 15 次，间隔 60 秒
        TxTrace trace = traceMapper.selectByTxId(txId);
        if (trace == null) {
            // 查不到不等于一定回滚：本地事务可能还在执行中。
            // 先看消息年龄，太新就回答未知，让 Broker 稍后再问。
            long ageMs = System.currentTimeMillis() - msg.getBornTimestamp();
            return ageMs > 60000
                    ? LocalTransactionState.ROLLBACK_MESSAGE
                    : LocalTransactionState.UNKNOW;
        }
        return trace.isCommitted()
                ? LocalTransactionState.COMMIT_MESSAGE
                : LocalTransactionState.ROLLBACK_MESSAGE;
    }
}
回查实现要点为什么必须这样做错的后果
本地事务内插入以事务 ID 为唯一键的留痕回查的唯一可靠依据只能回答未知，半消息超过回查次数后丢弃，下游静默少数据
回查方法只读，不做任何写入同一事务会被回查多次每次回查都改一次状态，业务数据被反复修改
查不到记录时先判断消息年龄本地事务可能还在执行中刚发出就回答回滚，本地成功但消息没了
留痕表保留期覆盖最大回查窗口回查最长可持续十几分钟留痕被提前清理，回查误判为回滚
留痕表与业务表同库跨库就不再是一个本地事务退化成又一个分布式事务问题

TCC：一张事务记录表解决三个问题

空回滚与悬挂的成因主干篇已经讲过，这里给实现。核心是所有分支操作都先读写同一张记录表，用唯一约束加状态字段把三类问题一次性收口。

CREATE TABLE `tcc_branch` (
  `id`           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `xid`          VARCHAR(64) NOT NULL COMMENT '全局事务 ID，贯穿日志与链路',
  `branch_id`    VARCHAR(64) NOT NULL COMMENT '分支 ID，通常是 xid 加资源标识',
  `resource`     VARCHAR(64) NOT NULL COMMENT '资源名，如 inventory / balance',
  `state`        TINYINT     NOT NULL COMMENT '1 已预留 2 已确认 3 已取消',
  `tried_at`     DATETIME(3) NULL,
  `confirmed_at` DATETIME(3) NULL,
  `cancelled_at` DATETIME(3) NULL,
  PRIMARY KEY (`id`),
  -- 唯一键同时承担幂等判定与空回滚判定，是三个问题的共同支点
  UNIQUE KEY `uk_branch` (`branch_id`),
  KEY `idx_xid` (`xid`),
  -- 巡检长时间停在已预留状态的分支
  KEY `idx_state_time` (`state`, `tried_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='TCC 分支事务记录';

@Transactional(rollbackFor = Exception.class)
public boolean tryDeduct(String xid, String branchId, long skuId, int qty) {
    // 防悬挂：若 Cancel 已先到过（记录为已取消），迟到的 Try 必须直接拒绝，
    // 否则这次预留将永远没人释放。
    TccBranch exist = branchMapper.selectForUpdate(branchId);
    if (exist != null) {
        return exist.getState() == State.TRIED; // 幂等：重复 Try 返回上次结果
    }
    branchMapper.insertTried(xid, branchId, "inventory"); // 唯一键保证并发只成功一次
    // 业务预留：冻结列加、可用列减，条件 UPDATE 保证不会预留出负数
    int rows = stockMapper.freeze(skuId, qty);            // WHERE available >= qty
    if (rows == 0) {
        throw new BizException("库存不足");                // 抛出后记录与业务一起回滚
    }
    return true;
}

@Transactional(rollbackFor = Exception.class)
public void cancelDeduct(String xid, String branchId, long skuId, int qty) {
    TccBranch exist = branchMapper.selectForUpdate(branchId);
    if (exist == null) {
        // 空回滚：Try 从未执行（超时或根本没到达），插入已取消记录后返回成功。
        // 这条插入同时为迟到的 Try 留下拒绝依据，把空回滚与防悬挂串成一套机制。
        branchMapper.insertCancelled(xid, branchId, "inventory");
        return;
    }
    if (exist.getState() == State.CANCELLED) {
        return; // 幂等：重复 Cancel 直接成功
    }
    stockMapper.unfreeze(skuId, qty);
    // 带前置状态的条件更新，防止 Confirm 与 Cancel 并发时互相覆盖
    branchMapper.casState(branchId, State.TRIED, State.CANCELLED);
}
问题触发条件这张表怎么解决
幂等协调者重试，同一分支被调多次唯一键加状态判断，重复调用直接返回上次结果
空回滚Try 超时未执行，协调者仍发 Cancel查不到记录则插入已取消并返回成功，不碰业务数据
悬挂Try 的请求在网络里延迟，在 Cancel 之后才到Try 发现已有已取消记录，直接拒绝执行
并发状态错乱Confirm 与 Cancel 几乎同时到达状态流转用带前置状态的条件更新，受影响行数为 0 即放弃
卡在中间态协调者崩溃，二阶段没人发按 idx_state_time 巡检长时间已预留的分支并推进

Saga 编排：状态机表与补偿失败兜底

Saga 的核心挑战不在正向流程，而在补偿失败。任何补偿操作都可能失败——下游宕机、网络分区、接口限流——不给它一个兜底链条，系统就会在中间态永久挂起。

CREATE TABLE `saga_instance` (
  `id`              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `flow_id`         VARCHAR(64)  NOT NULL COMMENT '流程实例 ID，也是全局事务 ID',
  `flow_type`       VARCHAR(32)  NOT NULL COMMENT '流程类型：如 CREATE_ORDER',
  `current_step`    INT          NOT NULL DEFAULT 0 COMMENT '当前步骤序号',
  `direction`       TINYINT      NOT NULL DEFAULT 1 COMMENT '1 正向 2 逆向补偿',
  `state`           VARCHAR(20)  NOT NULL COMMENT 'RUNNING/COMPENSATING/SUCCESS/COMP_FAIL/MANUAL',
  `steps_done`      VARCHAR(512) NOT NULL DEFAULT '' COMMENT '已完成步骤序号列表，逗号分隔',
  `comp_progress`   VARCHAR(512) NOT NULL DEFAULT '' COMMENT '已完成补偿序号列表',
  `context`         TEXT         NOT NULL COMMENT '流程上下文 JSON：各步骤的入参与出参快照',
  `retry_count`     INT          NOT NULL DEFAULT 0,
  `next_retry_at`   DATETIME(3)  NULL,
  `created_at`      DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  `updated_at`      DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_flow` (`flow_id`),
  KEY `idx_state_retry` (`state`, `next_retry_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Saga 编排状态机';
stateDiagram-v2 [*] --> Running Running --> Running: 当前步骤成功，推进到下一步 Running --> Success: 所有步骤完成 Running --> Compensating: 某步骤失败，开始逆向补偿 Compensating --> Compensating: 当前补偿成功，补偿上一步 Compensating --> CompFail: 某补偿步骤重试达上限 CompFail --> Manual: 进入人工介入队列 Manual --> Compensating: 人工修复后重新驱动补偿 Manual --> Success: 人工确认数据已修正 Compensating --> [*]: 所有已完成步骤补偿完毕 Success --> [*]
驱动方式做法适合代价
定时扫描按 idx_state_retry 扫描 RUNNING 和 COMPENSATING 的实例，驱动下一步吞吐量不高但需要绝对可靠的场景推进延迟受扫描间隔支配，典型 1 到 5 秒
事件驱动每步完成后发事件，消费者推进下一步，超时兜底用扫描高吞吐低延迟场景必须配合扫描兜底，否则消息丢了流程就停了

补偿失败的处置链条必须在设计时就定好，上线后再补来不及：

阶段做法关键约束
指数退避重试间隔 5s、30s、2m、10m，覆盖下游短时不可用重试次数上限 6 到 10 次
标记补偿失败超过重试上限，实例状态改为 COMP_FAIL状态变更后不再自动重试，等人工
告警带全局事务 ID、卡住步骤、失败原因、业务上下文摘要告警必须到人，不能只写日志
人工介入值班通过后台接口对该步骤发起重试或手动修数据重试调用的仍然是幂等补偿接口
绝不能静默丢弃没有「跳过这步继续」的选项，数据会永久不一致跳过只能由 DBA 在数据修正确认后操作

补偿操作的设计原则

让补偿尽可能不失败。比如「扣减库存」的补偿不是「把可用库存加回去」（UPDATE 有并发与负数风险），而是「插入一条返还流水」（INSERT 基本只有主键冲突，天然幂等且失败面极窄）。把高风险操作转成低风险操作，是降低补偿失败率的核心手段。

超时与中间态巡检：所有柔性事务的共同需求

无论用哪种方案，「长时间处于中间状态」的实例都必须有巡检任务兜底。巡检不是可选的锦上添花，是方案能不能在生产跑的前提。

-- 巡检本地消息表：投递中超过 3 分钟的记录，要么超时回收，要么告警人工
SELECT id, biz_id, locked_at, owner, retry_count
  FROM tx_message
 WHERE status = 1
   AND locked_at < DATE_SUB(NOW(3), INTERVAL 180 SECOND)
 LIMIT 200;

-- 巡检 TCC 分支：预留超过 10 分钟仍未推进到确认或取消
SELECT xid, branch_id, resource, tried_at
  FROM tcc_branch
 WHERE state = 1
   AND tried_at < DATE_SUB(NOW(3), INTERVAL 600 SECOND)
 LIMIT 200;

-- 巡检 Saga 实例：运行中或补偿中超过 30 分钟
SELECT flow_id, flow_type, state, current_step, updated_at
  FROM saga_instance
 WHERE state IN ('RUNNING', 'COMPENSATING')
   AND updated_at < DATE_SUB(NOW(3), INTERVAL 1800 SECOND)
 LIMIT 200;
巡检目标中间态含义处置策略
本地消息 status=1抢占后未成功投递或未回写状态超时回收改回待投递，若 retry_count 已达上限则转死信
TCC 分支 state=已预留协调者崩溃，二阶段未发出查询协调者侧的全局事务状态决定继续确认还是取消；查不到则超时后回滚
Saga state=RUNNING驱动任务丢了或下游无限等待重新驱动当前步骤，下游必须幂等
Saga state=COMPENSATING补偿驱动中断重新驱动当前补偿步骤
Saga state=COMP_FAIL补偿重试耗尽告警加人工介入，不自动处理

对账是最后一道防线

重试、补偿、巡检能覆盖大部分故障场景，但覆盖不了全部：代码 bug 导致状态机走错分支、巡检任务自己挂了一段时间、人工处理时改错了数据。这些漏网之鱼只有对账能发现。

-- 对账核心逻辑：两侧按时间范围汇总，找出差异
-- 左侧是订单库（源头），右侧是积分库（下游），按小时对齐

-- 1) 逐笔对比：以订单为基准，查积分侧是否有对应记录
SELECT o.order_id, o.amount, o.paid_at, p.points_id
  FROM orders o
  LEFT JOIN points_ledger p ON p.biz_ref = CONCAT('order:', o.order_id)
 WHERE o.paid_at BETWEEN '2026-09-23 10:00:00' AND '2026-09-23 11:00:00'
   AND o.status = 'PAID'
   AND p.points_id IS NULL;
-- 结果不为空 = 有订单已支付但积分未到账

-- 2) 汇总比对：对同一时间窗口的记录数与金额求和，大数上快速发现偏差
SELECT 'orders' AS side, COUNT(*) AS cnt, SUM(amount) AS total
  FROM orders
 WHERE paid_at BETWEEN '2026-09-23 10:00:00' AND '2026-09-23 11:00:00'
   AND status = 'PAID'
UNION ALL
SELECT 'points' AS side, COUNT(*) AS cnt, SUM(biz_amount) AS total
  FROM points_ledger
 WHERE created_at BETWEEN '2026-09-23 10:00:00' AND '2026-09-23 11:00:00';
对账设计要素核心决策建议
对什么逐笔明细 + 汇总金额 / 数量两层都做：汇总快速发现大面积问题，逐笔定位具体单据
多久对一次取决于业务容忍的不一致窗口核心资金链路按小时，一般业务按天，非关键可按周
发现差异后怎么办自动修复 vs 告警能明确判定方向且修复幂等的可自动补发；涉及资金的先告警，人工确认后修
修复的幂等性补发消息或重新调用下游用原始 biz_id 作为幂等键，保证补发不会产生重复数据
对账本身的留痕对了什么、什么结果、修了什么每次对账写一条汇总记录加差异明细，便于审计和追溯

可观测性：全局事务 ID 必须贯穿全链路

柔性事务必须能回答：「这笔业务现在卡在哪一步？」答不出来，故障时就只能翻日志猜。全局事务 ID 是贯穿所有组件的主线。

观测维度怎么做不做的后果
日志携带 xid写入 MDC / ThreadLocal，日志 pattern 自动输出故障时要在多个服务的日志里用业务主键一条一条搜，耗时几十分钟到几小时
消息传递 xid放在 MQ 消息 properties / header 里异步链路断链，消费端日志无法关联到原始请求
状态查询接口按 xid 查出所有分支记录与当前状态客服和运营无法自助查进度，所有卡单都升级到开发
指标看板中间态数量、平均停留时长、死信数、补偿失败数无法主动发现堆积，只能被动等投诉

反模式清单

反模式为什么出现实际后果正确做法
消息表不加 biz_id 唯一键只靠自增 ID 做主键业务重试产生两条消息，下游重复处理biz_id 唯一键，重复插入走冲突分支
投递任务不做抢占以为只部署一个实例就不需要扩容后立即出现重复投递一律用条件 UPDATE 抢占，单实例也不例外
补偿接口不幂等补偿只会调一次的假设重试触发时下游数据被反复修改补偿用条件更新或唯一键保护
没有超时巡检乐观地认为驱动任务不会挂驱动进程崩溃后流程永久停在中间态，到对账才发现所有中间态都有超时巡检兜底
没有对账重试和补偿够了的假设漏网不一致累积数月，靠用户投诉才暴露对账是最后一道防线，必须有
把中间态直接暴露给用户前端直接展示数据库状态字段用户看到「补偿中」「悬挂」等术语，恐慌并反复提单业务状态与内部状态分层映射
在本地事务里做远程调用想把远程调用包在数据库事务里保证原子长事务占住行锁和连接，把一致性问题换成了可用性问题远程调用移到事务之外，配本地消息表或事务消息保证最终一致
事务消息不留痕以为回查只是形式回查全部返回未知，半消息超限丢弃，下游静默少数据本地事务内写以事务 ID 为唯一键的留痕记录
❌
最容易踩的坑
  • 消息表没加唯一键：业务重试时插入两条，下游收到两条且无法区分，少一个唯一键就少了整个方案的地基。
  • 投递任务不做抢占：多实例同时扫到同一条消息各发一次，下游以为是两笔不同业务。
  • 回查接口改了数据：Broker 默认回查 15 次，每次都调一遍回查方法，如果回查里有写入，业务数据会被改 15 遍。
  • 本地事务没有为回查留痕：回查只能返回未知，半消息超过回查上限后被丢弃，表现为下游偶尔收不到消息，且集中在高峰期（本地事务慢的时候）。
  • 补偿接口不幂等：补偿被重试多次，每次都执行一遍，退款退了两倍、库存加回了两倍。
  • 补偿失败后静默跳过：流程标记成功但下游数据没改回来，不一致永久化，后续对账发现时已无法自动修复。
  • 没有超时巡检：驱动进程崩溃后所有进行中的流程永久停滞，到下一次对账（可能是明天）才会发现。
  • 退避序列起步太短：下游挂了 5 分钟，起步间隔 1 秒的重试会在 5 分钟里打出几百次无效调用，增大下游恢复难度。
  • 消息表不做归档：日均百万单跑三个月就是一亿行，扫描索引变高变慢，投递延迟从毫秒涨到秒级。
  • 全局事务 ID 没有传进异步链路：消息消费端的日志与上游完全断链，排查卡单只能按业务主键一条一条搜。
  • 把内部事务状态直接展示给用户：用户看到「已预留」「补偿失败」等术语，不知道自己的业务到底成了没有。
