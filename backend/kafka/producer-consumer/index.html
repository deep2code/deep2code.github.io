生产者与消费者

🏷️ Kafka,Go,Python,消息队列

客户端编程的误区

很多人写 Kafka 客户端代码的方式是"调个 API 发出去/拉回来"就完事了。但生产环境的客户端代码必须处理：重试语义、序列化兼容性、消费位点提交时机、Rebalance 监听器、死信处理、背压控制。我见过太多上线即事故的代码——生产端没开幂等导致重复扣款，消费端自动提交 Offset 导致宕机丢消息。本文用 sarama（Go）和 confluent-kafka（Python）两个最主流的客户端库，从生产者到消费者写出生产可用的代码。

生产者核心机制

生产者不是"发一条消息就开一个网络请求"——那样性能极差。内部有一个 RecordAccumulator（记录累加器），按 Partition 维护一个双端队列。Producer 把消息先丢进队列，后台 Sender 线程把同一 Partition 的消息攒成批次（Batch），满足 batch.size 或 linger.ms 任一条件就发送。这就是 Kafka 高吞吐的客户端侧实现。

消费者核心机制

消费者是拉模式：在一个循环里不断调用 poll()，Broker 返回一批消息。处理完后提交 Offset。消费组发生 Rebalance 时（成员加入/退出），Partition 会在消费者之间重新分配——这是必须处理的复杂场景。

Offset 提交时机的坑：我踩过最严重的一个坑就是自动提交。enable.auto.commit=true 默认每 5 秒提交一次，意味着你 poll 到消息后还没处理完，Offset 可能已经提交了。一旦进程崩溃，这批消息就丢了。生产环境务必关闭自动提交，处理完业务逻辑再手动提交。

Go 生产者实战（sarama）

package main

import (
    "log"
    "os"
    "os/signal"
    "syscall"
    "time"

    "github.com/IBM/sarama"
)

// OrderEvent 业务消息结构体会被序列化为 JSON
type OrderEvent struct {
    OrderID   string    `json:"order_id"`
    UserID    string    `json:"user_id"`
    Amount    float64   `json:"amount"`
    Status    string    `json:"status"`
    Timestamp time.Time `json:"timestamp"`
}

func newAsyncProducer(brokers []string) (sarama.AsyncProducer, error) {
    config := sarama.NewConfig()

    // === 可靠性配置 ===
    config.Producer.RequiredAcks = sarama.WaitForAll // 所有 ISR 确认才算成功
    config.Producer.Retry.Max = 5                     // 重试 5 次
    config.Producer.Retry.Backoff = 100 * time.Millisecond
    config.Producer.Idempotent = true                 // 幂等（避免重试导致重复）
    config.Net.MaxOpenRequests = 1                    // 幂等要求单连接

    // === 批量性能配置 ===
    config.Producer.Flush.Bytes = 64 * 1024           // 64KB 批次
    config.Producer.Flush.Messages = 1000             // 或攒够 1000 条
    config.Producer.Flush.Frequency = 500 * time.Millisecond

    // === 压缩 ===
    config.Producer.Compression = sarama.CompressionSnappy // 高吞吐用 lz4
    config.Producer.CompressionLevel = -1

    // === 元数据 ===
    config.Metadata.Full = true
    config.Metadata.Timeout = 10 * time.Second

    return sarama.NewAsyncProducer(brokers, config)
}

func main() {
    brokers := []string{"node-1:9092", "node-2:9092", "node-3:9092"}

    producer, err := newAsyncProducer(brokers)
    if err != nil {
        log.Fatalf("创建生产者失败: %v", err)
    }
    defer producer.Close()

    // 异步处理成功/失败回调
    go func() {
        for success := range producer.Successes() {
            log.Printf("发送成功: topic=%s partition=%d offset=%d",
                success.Topic, success.Partition, success.Offset)
        }
    }()

    go func() {
        for err := range producer.Errors() {
            log.Printf("发送失败: topic=%s err=%v", err.Msg.Topic, err.Err)
            // 实际生产中这里要写入本地磁盘或发往死信 Topic
        }
    }()

    // 发送 100 条订单事件
    for i := 0; i < 100; i++ {
        event := OrderEvent{
            OrderID:   fmt.Sprintf("ORD-%d", i),
            UserID:    fmt.Sprintf("user-%d", i%10),
            Amount:    99.50,
            Status:    "CREATED",
            Timestamp: time.Now(),
        }

        value, _ := json.Marshal(event)

        producer.Input() <- &sarama.ProducerMessage{
            Topic: "orders",
            Key:   sarama.StringEncoder(event.OrderID),
            Value: sarama.ByteEncoder(value),
            Headers: []sarama.RecordHeader{
                {Key: []byte("trace-id"), Value: []byte(uuid.New().String())},
            },
        }
    }

    // 优雅退出
    sigchan := make(chan os.Signal, 1)
    signal.Notify(sigchan, syscall.SIGINT, syscall.SIGTERM)
    <-sigchan
    producer.AsyncClose()
    log.Println("生产者已关闭")
}

Go 消费者实战（sarama）

package main

import (
    "context"
    "log"
    "os"
    "os/signal"
    "syscall"
    "time"

    "github.com/IBM/sarama"
)

// ConsumerGroupHandler 实现 sarama.ConsumerGroupHandler 接口
type OrderHandler struct {
    ready chan bool
}

func (h *OrderHandler) Setup(sarama.ConsumerGroupSession) error {
    close(h.ready)
    return nil
}

func (h *OrderHandler) Cleanup(sarama.ConsumerGroupSession) error {
    return nil
}

func (h *OrderHandler) ConsumeClaim(session sarama.ConsumerGroupSession, claim sarama.ConsumerGroupClaim) error {
    for msg := range claim.Messages() {
        log.Printf("收到消息: topic=%s partition=%d offset=%d key=%s",
            msg.Topic, msg.Partition, msg.Offset, string(msg.Key))

        // === 业务处理 ===
        var event OrderEvent
        if err := json.Unmarshal(msg.Value, &event); err != nil {
            log.Printf("反序列化失败, 跳过: %v", err)
            session.MarkMessage(msg, "") // 仍然提交，避免毒丸消息阻塞
            continue
        }

        if err := processOrder(&event); err != nil {
            log.Printf("处理失败: %v", err)
            // 实际生产中这里应该发往 retry topic 或 DLQ
            continue // 不提交 Offset，下次会重新消费
        }

        // 手动提交 Offset（处理成功后再提交）
        session.MarkMessage(msg, "")
    }
    return nil
}

func main() {
    brokers := []string{"node-1:9092", "node-2:9092", "node-3:9092"}

    config := sarama.NewConfig()
    config.Consumer.Group.Rebalance.Strategy = sarama.NewBalanceStrategySticky()
    config.Consumer.Group.Session.Timeout = 20 * time.Second
    config.Consumer.Group.Heartbeat.Interval = 3 * time.Second
    config.Consumer.Offsets.Initial = sarama.OffsetNewest
    config.Consumer.Offsets.AutoCommit.Enable = false      // 关闭自动提交，手动控制
    config.Consumer.MaxProcessingTime = 100 * time.Millisecond

    handler := &OrderHandler{ready: make(chan bool)}

    consumerGroup, err := sarama.NewConsumerGroup(brokers, "order-processor-group", config)
    if err != nil {
        log.Fatal(err)
    }
    defer consumerGroup.Close()

    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()

    go func() {
        for {
            if err := consumerGroup.Consume(ctx, []string{"orders"}, handler); err != nil {
                log.Printf("消费错误: %v", err)
            }
            if ctx.Err() != nil {
                return
            }
            handler.ready = make(chan bool)
        }
    }()

    <-handler.ready
    log.Println("消费者已启动，等待消息...")

    sigchan := make(chan os.Signal, 1)
    signal.Notify(sigchan, syscall.SIGINT, syscall.SIGTERM)
    <-sigchan
    log.Println("消费者已关闭")
}

Python 生产者（confluent-kafka）

import json
import time
import logging
from confluent_kafka import Producer

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

config = {
    'bootstrap.servers': 'node-1:9092,node-2:9092,node-3:9092',
    'acks': 'all',                          # 所有 ISR 确认
    'retries': 5,
    'retry.backoff.ms': 100,
    'enable.idempotence': True,             # 幂等生产者
    'compression.type': 'snappy',
    'linger.ms': 10,                        # 攒 10ms 再发，提升批量效率
    'batch.size': 65536,                    # 64KB
    'queue.buffering.max.messages': 100000,
    'delivery.timeout.ms': 30000,
}

producer = Producer(config)

def delivery_report(err, msg):
    # 消息投递回调（异步）
    if err is not None:
        logger.error(f'消息投递失败: {err}')
    else:
        logger.debug(f'消息投递成功: partition={msg.partition()} offset={msg.offset()}')

for i in range(100):
    event = {
        'order_id': f'ORD-{i}',
        'user_id': f'user-{i % 10}',
        'amount': 99.50,
        'status': 'CREATED',
        'timestamp': int(time.time()),
    }

    producer.produce(
        topic='orders',
        key=event['order_id'].encode(),
        value=json.dumps(event).encode(),
        on_delivery=delivery_report,
        headers=[('trace-id', f'trace-{i}')],
    )

    # 非阻塞地处理事件回调
    producer.poll(0)

# 等待所有消息发送完成
producer.flush(timeout=10)
logger.info('所有消息已发送完成')

Python 消费者（confluent-kafka）

import json
import signal
import logging
from confluent_kafka import Consumer, KafkaError

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

running = True

def signal_handler(sig, frame):
    global running
    running = False
    logger.info('收到退出信号，准备关闭消费者...')

signal.signal(signal.SIGINT, signal_handler)
signal.signal(signal.SIGTERM, signal_handler)

config = {
    'bootstrap.servers': 'node-1:9092,node-2:9092,node-3:9092',
    'group.id': 'order-processor-group',
    'auto.offset.reset': 'earliest',
    'enable.auto.commit': False,
    'max.poll.records': 500,
    'session.timeout.ms': 30000,
    'heartbeat.interval.ms': 10000,
    'fetch.min.bytes': 1024,
    'fetch.max.wait.ms': 5000,
}

consumer = Consumer(config)
consumer.subscribe(['orders'])

try:
    while running:
        msg = consumer.poll(timeout=1.0)

        if msg is None:
            continue

        if msg.error():
            if msg.error().code() == KafkaError._PARTITION_EOF:
                logger.info(f"分区到达末尾: {msg.topic()}[{msg.partition()}]")
            else:
                logger.error(f"消费错误: {msg.error()}")
            continue

        # === 业务处理 ===
        try:
            event = json.loads(msg.value().decode())
            logger.info(f"处理订单: {event['order_id']}")

            # TODO: 实际业务逻辑

            # 处理成功后手动提交 Offset
            consumer.commit(message=msg, asynchronous=False)

        except json.JSONDecodeError as e:
            logger.error(f"反序列化失败: {e}, 消息: {msg.value()!r}")
            # 毒丸消息也提交，避免无限重试
            consumer.commit(message=msg, asynchronous=False)

        except Exception as e:
            logger.error(f"处理失败: {e}, 不提交 Offset 等待重试")
            # 不 commit，下次 poll 会重新拉到这条消息

finally:
    consumer.close()
    logger.info('消费者已优雅关闭')

生产者侧

  • 必开幂等：enable.idempotence=true + acks=all，让 Broker 去重，避免重试导致重复消息。
  • 合理设置 linger.ms：1~20ms 是常见的平衡点。设为 0 延迟低但批量小，吞吐差。
  • 选择合适的压缩算法：CPU 富余用 zstd（压缩率最高），追求速度用 lz4，通用场景 snappy。
  • 缓冲区监控：如果 queue.buffering.max.messages 打满，说明 Broker 跟不上，需要扩容或排查下游瓶颈。

消费者侧

  • 关闭自动提交：enable.auto.commit=false，处理完业务再手动提交。自动提交可能导致"消息还没处理就提交了 Offset"，宕机后消息丢失。
  • 控制 poll 间隔：处理时间不能超过 max.poll.interval.ms（默认 5 分钟）。长耗时任务应该把消息丢到工作池，主循环快速返回。
  • 处理 Rebalance：Rebalance 期间所有消费者停止消费。实现 Rebalance 监听器，在 Rebalance 前提交未完成的 Offset，避免重复消费。
  • 消费组数 = Partition 数：如果消费组内消费者数超过 Partition 数，多出来的消费者空转。不要盲目扩容消费者。

常见问题 FAQ

Q1：如何保证消息的严格顺序？

Kafka 只保证单个 Partition 内的消息有序。要保证业务维度的顺序（比如同一个用户的事件有序），就把该业务 ID 作为消息 key，相同 key 的消息路由到同一个 Partition。不要试图用单 Partition 实现全局有序——那会彻底丧失水平扩展能力。如果业务真的需要全局有序，说明你的架构有问题，应该换一个更合适的中间件。

Q2：消费者 Rebalance 频繁怎么办？

Rebalance 频繁的常见原因：(1) 消息处理太慢，超过 max.poll.interval.ms，被 coordinator 踢出去；(2) GC 停顿导致心跳超时；(3) 网络抖动。解决方案：增大 session.timeout.ms（到 30s）和 max.poll.interval.ms（到 300s），减少每次 poll 的消息数（max.poll.records=100），用 CooperativeSticky 协议代替旧的 Eager 协议——增量 Rebalance 只迁移需要变化的 Partition，不全量 stop-the-world。

Q3：处理消息时崩溃了，Offset 没提交，重启后会重复消费吗？

会的，这是 Kafka 至少一次投递语义的代价。解决重复消费的方案：幂等消费。把业务唯一 ID（如 OrderID）存在数据库或 Redis 里，处理前先查一下是否已处理过。或者利用 Kafka 事务 API（可靠性篇详解）实现端到端 Exactly-Once。大多数场景下幂等消费就够了，不需要上重型事务。

总结

写 Kafka 客户端代码的关键不是"发消息"和"收消息"这两个 API，而是理解批量机制、Offset 提交时机、Rebalance 处理、幂等设计这四个容易踩坑的地方。Go 生态推荐 sarama（成熟稳定），Python 推荐 confluent-kafka（librdkafka 绑定，性能最好）。生产环境的客户端配置比代码本身更重要——本文中的每个配置项都值得你花时间理解背后的含义。
