FreeType

package main

import (
 "fmt"
 "image"
 "image/draw"
 "image/png"
 "os"
 "github.com/golang/freetype"
)

func DrawText(text string) {
 data, err := os.ReadFile("/System/Library/Fonts/STHeiti Medium.ttc")
 if err != nil {
  panic(err)
 }
 f, err := freetype.ParseFont(data)
 if err != nil {
  panic(err)
 }

 dst := image.NewRGBA(image.Rect(0, 0, 800, 600))
 draw.Draw(dst, dst.Bounds(), image.White, image.Point{}, draw.Src)

 c := freetype.NewContext()
 c.SetDst(dst)
 c.SetClip(dst.Bounds())
 c.SetSrc(image.Black)
 c.SetFont(f)
 fontSize := float64(50)
 // 字体越大, 显示越大
 c.SetFontSize(fontSize)

 // Pt是控制起点,Pt{x,y},x表示左起点,y表示下起点,y-fontSize才是上起点
 _, err = c.DrawString(text, freetype.Pt(0, int(fontSize)))
 if err != nil {
  panic(err)
 }

 pngFile, err := os.Create("draw.png")
 if err != nil {
  panic(err)
 }
 defer pngFile.Close()

 err = png.Encode(pngFile, dst)
 if err != nil {
  panic(err)
 }
}

func main() {
 DrawText("中国人golang语言教程ABC122")
}

FreeType 更新

Go 的 freetype 库主要用于字体渲染。到 2026 年，Go 侧字体渲染的事实标准已彻底转移到官方媒体库 golang.org/x/image：原 github.com/golang/freetype 仓库已进入只读归档状态（GitHub archived），最后一次代码推送停留在 2024 年，新项目不应再以它为依赖锚点。

2026 年的推荐字体链路

golang.org/x/image 仍是活跃项目，2026-08-11 发布 v0.45.0，仓库保持月度级维护提交。官方推荐的组合是：

  • golang.org/x/image/font：字面抽象（Face 接口）与绘制入口（Drawer）；
  • golang.org/x/image/font/sfnt：TTF/OTF 字体文件解析，2026 年仍持续修复安全边界问题；
  • golang.org/x/image/font/opentype：OpenType 字型光栅化，同时支持 TTF 与 OTF/CFF 双轮廓；
  • golang.org/x/image/font/gofont：内置 Go 字体族，免去打包字体文件的麻烦。

相比老 github.com/golang/freetype 只支持 TrueType 轮廓与 Unicode 编码的历史包袱，上面这条链路已覆盖 OTF/CFF 等现代字体形态。

迁移与工程建议

老代码里 ioutil.ReadFile("/System/Library/Fonts/STHeiti Medium.ttc") 这类硬编码系统字体路径在新项目里应当避免：一是跨平台不可移植，二是 .ttc 集合字体需要先解析再按 index 取出目标字形表。建议字体文件用 go:embed 打进二进制，或先用 font/sfnt 解析后按指定 index 加载。

另外要明确边界：x/image 主打原生光栅化，不做 HarfBuzz 级别的复杂文字整形（阿拉伯文连字、藏文重排等）。这类需求在 Go 生态里的成熟选择是 github.com/go-text/typesetting——纯 Go 实现、2026 年仍活跃维护，Fyne、Gio、Ebitengine 等 GUI 框架都在使用。纯中文 / 拉丁文本场景用 x/image 就够了。
