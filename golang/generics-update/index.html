Go 泛型进阶（1.18-1.27）

Go 1.18 引入泛型（Type Parameters），这是 Go 语言自 1.0 以来最大的语法变更。1.27 引入了泛型方法（Generic Methods），方法声明可以有自己的类型参数。

泛型最新特性

Go 1.23 迭代器

Go 1.23 引入了 range over func 特性，函数可以作为迭代器使用：

// 迭代器函数类型
func(yield func(int) bool)

// 使用示例
func Squares() func(yield func(int) bool) {
    return func(yield func(int) bool) {
        for i := 1; i <= 10; i++ {
            if !yield(i * i) {
                return
            }
        }
    }
}

// 使用
for sq := range Squares() {
    fmt.Println(sq)  // 1, 4, 9, 16, ...
}

Go 1.24 泛型类型别名

泛型类型别名允许更灵活的类型抽象：

// 泛型类型别名
type Map[K any, V any] = map[K]V

// 泛型别名使用
func Process[K comparable, V any](m Map[K]V) {
    for k, v := range m {
        fmt.Println(k, v)
    }
}

Go 1.27 泛型方法

方法声明可以有自己的类型参数：

// 泛型方法
type Container[T any] struct {
    value T
}

func (c Container[T]) Get() T {
    return c.value
}

// 泛型方法带约束
func (c Container[T]) Transform(func(T) T) Container[T] {
    return Container[T]{value: func(c.value)}
}

// 使用泛型方法
c := Container[int]{value: 42}
result := c.Transform(func(x int) int { return x * 2 })

泛型基础

类型参数

// 泛型函数
func PrintSlice[T any](s []T) {
    for _, v := range s {
        fmt.Println(v)
    }
}

// 使用
PrintSlice([]int{1, 2, 3})
PrintSlice([]string{"a", "b", "c"})

// 泛型结构体
type Pair[K any, V any] struct {
    Key   K
    Value V
}

p := Pair[string, int]{Key: "age", Value: 25}

类型约束

// 使用接口作为约束
type Number interface {
    int | int32 | int64 | float32 | float64
}

func Add[T Number](a, b T) T {
    return a + b
}

// 使用 cmp.Ordered (Go 1.21+)
func Max[T cmp.Ordered](a, b T) T {
    if a > b {
        return a
    }
    return b
}

// 使用 any 约束（无限制）
func Identity[T any](t T) T {
    return t
}

高级特性

泛型接口

// 泛型接口
type Container[T any] interface {
    Get() T
    Set(T)
}

// 泛型接口实现
type Box[T any] struct {
    value T
}

func (b Box[T]) Get() T {
    return b.value
}

func (b *Box[T]) Set(v T) {
    b.value = v
}

// 使用泛型接口
func Put[T any](c Container[T], v T) {
    c.Set(v)
}

泛型方法

// 泛型方法（Go 1.27+）
type Stack[T any] struct {
    items []T
}

func (s *Stack[T]) Push(item T) {
    s.items = append(s.items, item)
}

func (s *Stack[T]) Pop() (T, bool) {
    if len(s.items) == 0 {
        var zero T
        return zero, false
    }
    item := s.items[len(s.items)-1]
    s.items = s.items[:len(s.items)-1]
    return item, true
}

// 泛型方法带约束
func (s *Stack[T]) Find(predicate func(T) bool) *T {
    for _, item := range s.items {
        if predicate(item) {
            return &item
        }
    }
    return nil
}

类型推断

// 类型推断
func Map[T any, U any](s []T, fn func(T) U) []U {
    result := make([]U, len(s))
    for i, v := range s {
        result[i] = fn(v)
    }
    return result
}

// 完整类型参数
result1 := Map[int, string]([]int{1, 2, 3}, func(i int) string {
    return strconv.Itoa(i)
})

// 推断类型参数
result2 := Map([]int{1, 2, 3}, func(i int) string {
    return strconv.Itoa(i)
})

设计模式

Option 模式

// 泛型 Option 模式
type Option[T any] func(*T)

func WithTimeout[T any](timeout time.Duration) Option[T] {
    return func(t *T) {
        // 设置超时
    }
}

func NewServer[T any](opts ...Option[T]) *T {
    var t T
    for _, opt := range opts {
        opt(&t)
    }
    return &t
}

// 使用
server := NewServer(
    WithTimeout(5 * time.Second),
    WithPort(8080),
)

Result 模式

// Result 类型
type Result[T any] struct {
    Value T
    Err   error
}

func (r Result[T]) Ok() (T, error) {
    return r.Value, r.Err
}

func (r Result[T]) Must() T {
    if r.Err != nil {
        panic(r.Err)
    }
    return r.Value
}

// Try 函数
func Try[T any](fn func() (T, error)) Result[T] {
    v, err := fn()
    return Result[T]{Value: v, Err: err}
}

// 使用
result := Try(func() (int, error) {
    return 42, nil
})
value, err := result.Ok()

Builder 模式

// 泛型 Builder
type Builder[T any] struct {
    value T
}

func NewBuilder[T any](initial T) *Builder[T] {
    return &Builder[T]{value: initial}
}

func (b *Builder[T]) With(fn func(T) T) *Builder[T] {
    b.value = fn(b.value)
    return b
}

func (b *Builder[T]) Build() T {
    return b.value
}

// 使用
config := NewBuilder(Config{}).
    With(func(c Config) Config {
        c.Port = 8080
        return c
    }).
    With(func(c Config) Config {
        c.Debug = true
        return c
    }).
    Build()

最佳实践

何时使用泛型

  • 需要为多种类型编写相同逻辑时
  • 实现数据结构（栈、队列、链表等）时
  • 避免代码重复，提高类型安全
  • 不要过度使用：简单场景不必用泛型

约束最佳实践

  • 使用标准库约束（comparable、cmp.Ordered）
  • 自定义约束应使用小接口
  • 避免约束过于宽松导致运行时 panic

性能考虑

  • 泛型在编译时单态化，无运行时开销
  • 但可能导致二进制增大（代码膨胀）
  • 使用 go build -gcflags=-S 查看生成的代码
  • 性能敏感场景注意泛型实例化数量

实战示例

slices 和 maps 包

import (
    "cmp"
    "slices"
    "maps"
)

// slices 包
numbers := []int{3, 1, 4, 1, 5, 9, 2, 6}
slices.Sort(numbers)
slices.SortFunc(numbers, func(a, b int) int {
    return cmp.Compare(a, b)
})
slices.Reverse(numbers)

// 泛型查找
idx := slices.Index(numbers, 5)
found := slices.Contains(numbers, 5)

// maps 包
m := map[string]int{"a": 1, "b": 2, "c": 3}
keys := maps.Keys(m)
values := maps.Values(m)
maps.Copy(m, map[string]int{"d": 4})

// 泛型比较
equal := maps.Equal(m, map[string]int{"a": 1, "b": 2, "c": 3})

Channel 模式

// 泛型 Channel
func FanIn[T any](chs ...<-chan T) <-chan T {
    result := make(chan T)
    var wg sync.WaitGroup

    for _, ch := range chs {
        wg.Add(1)
        go func(c <-chan T) {
            defer wg.Done()
            for v := range c {
                result <- v
            }
        }(ch)
    }

    go func() {
        wg.Wait()
        close(result)
    }()

    return result
}

// 使用
ch1 := make(chan int)
ch2 := make(chan int)
merged := FanIn(ch1, ch2)

Context 模式

// 泛型 Context Value
func GetUser[T any](ctx context.Context, key string) (T, bool) {
    value := ctx.Value(key)
    if value == nil {
        var zero T
        return zero, false
    }
    v, ok := value.(T)
    return v, ok
}

// 使用
ctx := context.WithValue(context.Background(), "user", User{Name: "Alice"})
user, ok := GetUser[User](ctx, "user")
