package main

import (
    "fmt"
    "context"
    "time"
)

// 结构体
type User struct {
    ID   int    `json:"id"`
    Name string `json:"name"`
    Age  int    `json:"age,omitempty"`
}

// 接口
type Reader interface {
    Read(p []byte) (n int, err error)
}

// 函数
func greet(name string) string {
    return fmt.Sprintf("Hello, %s!", name)
}

// 方法
func (u User) String() string {
    return fmt.Sprintf("%s (%d)", u.Name, u.Age)
}

// 错误处理
func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, fmt.Errorf("division by zero")
    }
    return a / b, nil
}

// 泛型（Go 1.18+）
func Map[T, U any](slice []T, fn func(T) U) []U {
    result := make([]U, len(slice))
    for i, v := range slice {
        result[i] = fn(v)
    }
    return result
}

func main() {
    // 基本类型
    var i int = 42
    s := "hello"           // 短声明
    b := true
    
    // 切片和映射
    nums := []int{1, 2, 3}
    m := map[string]int{"a": 1, "b": 2}
    
    // 打印
    fmt.Printf("i=%d, s=%s, b=%v\n", i, s, b)
    fmt.Println(greet("World"))
    
    // 错误处理
    result, err := divide(10, 0)
    if err != nil {
        fmt.Println("Error:", err)
    } else {
        fmt.Println("Result:", result)
    }
    
    // 泛型使用
    ints := []int{1, 2, 3}
    strs := Map(ints, func(i int) string {
        return fmt.Sprintf("%d", i)
    })
    fmt.Println(strs)
}

并发编程

package main

import (
    "fmt"
    "sync"
    "time"
)

// WaitGroup 等待一组 goroutine 完成
func worker(id int, wg *sync.WaitGroup) {
    defer wg.Done()
    fmt.Printf("Worker %d starting\n", id)
    time.Sleep(time.Second)
    fmt.Printf("Worker %d done\n", id)
}

// Mutex 互斥锁
type Counter struct {
    mu    sync.Mutex
    value int
}

func (c *Counter) Increment() {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.value++
}

func (c *Counter) Value() int {
    c.mu.Lock()
    defer c.mu.Unlock()
    return c.value
}

// Channel 通信
func fanIn(ch1, ch2 <-chan string) <-chan string {
    ch := make(chan string)
    go func() {
        for {
            select {
            case s := <-ch1:
                ch <- s
            case s := <-ch2:
                ch <- s
            }
        }
    }()
    return ch
}

// Context 取消
func longRunningTask(ctx context.Context) error {
    select {
    case <-time.After(2 * time.Second):
        return nil
    case <-ctx.Done():
        return ctx.Err()
    }
}

func main() {
    // 启动 goroutine
    go func() {
        fmt.Println("Async operation")
    }()
    
    // WaitGroup
    var wg sync.WaitGroup
    for i := 1; i <= 3; i++ {
        wg.Add(1)
        go worker(i, &wg)
    }
    wg.Wait()
    
    // Mutex
    counter := &Counter{}
    var wg2 sync.WaitGroup
    for i := 0; i < 1000; i++ {
        wg2.Add(1)
        go func() {
            defer wg2.Done()
            counter.Increment()
        }()
    }
    wg2.Wait()
    fmt.Println("Counter:", counter.Value())
    
    // Channel
    ch1 := make(chan string)
    ch2 := make(chan string)
    go func() { ch1 <- "from 1" }()
    go func() { ch2 <- "from 2" }()
    
    result := <-fanIn(ch1, ch2)
    fmt.Println("Received:", result)
    
    // Context
    ctx, cancel := context.WithTimeout(context.Background(), time.Second)
    defer cancel()
    if err := longRunningTask(ctx); err != nil {
        fmt.Println("Task failed:", err)
    } else {
        fmt.Println("Task completed")
    }
}

错误处理

package main

import (
    "errors"
    "fmt"
)

// 自定义错误
var ErrNotFound = errors.New("resource not found")

type NotFoundError struct {
    Resource string
}

func (e *NotFoundError) Error() string {
    return fmt.Sprintf("%s not found", e.Resource)
}

// 错误包装
func readFile(path string) ([]byte, error) {
    data, err := internalRead(path)
    if err != nil {
        return nil, fmt.Errorf("failed to read %s: %w", path, err)
    }
    return data, nil
}

func internalRead(path string) ([]byte, error) {
    return nil, ErrNotFound
}

// 错误检查
func checkError(err error) {
    if errors.Is(err, ErrNotFound) {
        fmt.Println("Not found error")
    }
    
    var nfErr *NotFoundError
    if errors.As(err, &nfErr) {
        fmt.Printf("Specific error: %s\n", nfErr.Resource)
    }
}

func main() {
    _, err := readFile("test.txt")
    if err != nil {
        checkError(err)
        fmt.Println("Error:", err)
    }
}

Web 框架示例

package main

import (
    "net/http"
    "github.com/gin-gonic/gin"
    "github.com/go-playground/validator/v10"
)

// 结构体绑定
type User struct {
    Name  string `json:"name" validate:"required,min=2,max=50"`
    Email string `json:"email" validate:"required,email"`
    Age   int    `json:"age" validate:"gte=0,lte=150"`
}

var validate = validator.New()

func main() {
    r := gin.Default()
    
    // 中间件
    r.Use(func(c *gin.Context) {
        c.Set("request_id", "123")
        c.Next()
    })
    
    // 路由组
    api := r.Group("/api/v1")
    {
        api.GET("/health", func(c *gin.Context) {
            c.JSON(http.StatusOK, gin.H{"status": "ok"})
        })
        
        users := api.Group("/users")
        {
            users.GET("", getUsers)
            users.GET("/:id", getUser)
            users.POST("", createUser)
            users.PUT("/:id", updateUser)
            users.DELETE("/:id", deleteUser)
        }
    }
    
    r.Run(":8080") // 监听 0.0.0.0:8080
}

func getUsers(c *gin.Context) {
    c.JSON(http.StatusOK, gin.H{
        "users": []User{
            {Name: "Alice", Email: "alice@example.com", Age: 30},
            {Name: "Bob", Email: "bob@example.com", Age: 25},
        },
    })
}

func getUser(c *gin.Context) {
    id := c.Param("id")
    c.JSON(http.StatusOK, gin.H{"id": id, "name": "Alice"})
}

func createUser(c *gin.Context) {
    var user User
    if err := c.ShouldBindJSON(&user); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    if err := validate.Struct(user); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusCreated, gin.H{"user": user})
}

func updateUser(c *gin.Context) {
    id := c.Param("id")
    var user User
    if err := c.ShouldBindJSON(&user); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    c.JSON(http.StatusOK, gin.H{"id": id, "user": user})
}

func deleteUser(c *gin.Context) {
    id := c.Param("id")
    c.JSON(http.StatusOK, gin.H{"message": "deleted", "id": id})
}

测试

package main

import (
    "testing"
    "github.com/stretchr/testify/assert"
)

// 单元测试
func TestAdd(t *testing.T) {
    result := Add(2, 3)
    assert.Equal(t, 5, result, "2+3 should equal 5")
}

// 表格驱动测试
func TestAddTable(t *testing.T) {
    tests := []struct {
        name     string
        a, b     int
        expected int
    }{
        {"positive", 2, 3, 5},
        {"negative", -1, -1, -2},
        {"zero", 0, 0, 0},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result := Add(tt.a, tt.b)
            assert.Equal(t, tt.expected, result)
        })
    }
}

// Benchmark 基准测试
func BenchmarkAdd(b *testing.B) {
    for i := 0; i < b.N; i++ {
        Add(1, 2)
    }
}

// Example 示例测试（输出需要匹配）
func ExampleAdd() {
    sum := Add(1, 2)
    println(sum)
    // Output: 3
}

func Add(a, b int) int {
    return a + b
}

构建与部署

# 编译
go build -o myapp main.go           # 编译为可执行文件
go build -o myapp -tags prod main.go # 条件编译

# 测试
go test -v ./...                     # 运行所有测试
go test -cover ./...                 # 覆盖率
go test -bench=. -benchmem ./...    # 基准测试

# 静态分析
go vet ./...                         # 代码检查
golangci-lint run                    # 综合 linter

# 依赖管理
go mod init github.com/user/repo    # 初始化
go mod tidy                         # 整理依赖
go mod download                     # 下载依赖

# 交叉编译
GOOS=linux GOARCH=amd64 go build main.go   # Linux
GOOS=windows GOARCH=amd64 go build main.go # Windows
GOOS=darwin GOARCH=amd64 go build main.go  # macOS
