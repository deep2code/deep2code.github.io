func GroupBy[T any, U comparable](collection []T, iteratee func(T) U) map[U][]T {
 result := map[U][]T{}

 for _, item := range collection {
  key := iteratee(item)

  result[key] = append(result[key], item)
 }

 return result
}

切片分组更新

Go 1.21 的 slices 和 maps 包简化了分组操作：

import "slices"
import "maps"

// 泛型分组函数
func GroupBy[T any, K comparable](items []T, keyFn func(T) K) map[K][]T {
    result := make(map[K][]T)
    for _, item := range items {
        k := keyFn(item)
        result[k] = append(result[k], item)
    }
    return result
}

// 使用
type Order struct{ ID string; Status string }
orders := []Order{{"1", "paid"}, {"2", "pending"}, {"3", "paid"}}
grouped := GroupBy(orders, func(o Order) string { return o.Status })
// grouped["paid"] = [{1 paid}, {3 paid}]
// grouped["pending"] = [{2 pending}]

Go 1.23+：迭代器版的 group by

Go 1.23（2024-08）正式启用 range-over-func，并给 slices / maps 补齐了整套迭代器 API：slices.Values / All / Backward 产生迭代器，Collect / AppendSeq 收回切片，Chunk 直接原地分块为等长子切片；map 侧有 maps.Keys / Values / All 与 maps.Collect。分组可以写成「切片 → 迭代器 → map」的流式风格：

// Go 1.23+：迭代器风格
import "iter"
import "slices"

func GroupBySeq[T any, K comparable](seq iter.Seq[T], keyFn func(T) K) map[K][]T {
    out := make(map[K][]T)
    for v := range seq {
        k := keyFn(v) // 只算一次 key,别在 append 里重复调用函数
        out[k] = append(out[k], v)
    }
    return out
}

// 切片通过 Values 转成迭代器传入
groups := GroupBySeq(slices.Values(orders), func(o Order) string { return o.Status })

泛型方法时代的选型

Go 1.27（2026-08-19 发布）正式支持泛型方法后，分组逻辑还可以挂在具体类型上，语义更贴近「往数据结构上加方法」：

// Go 1.27+：泛型方法
type Items[T any] []T

func (s Items[T]) GroupBy[K comparable](keyFn func(T) K) map[K][]T {
    out := make(map[K][]T)
    for _, v := range s {
        out[keyFn(v)] = append(out[keyFn(v)], v)
    }
    return out
}

选型建议：存量手写泛型版本（上面的 1.21 写法）完全兼容、不用动；新代码按场景取舍——数据量小图省事用迭代器写法，热路径且数据量大时保留手写循环（迭代器封装有一层间接开销）；samber/lo 这类第三方泛型工具库（2026 年仍在活跃发版）可作备选，但标准库已经覆盖绝大多数分组场景。
