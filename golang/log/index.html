package main

import (
 "fmt"
 "io"
 "io"
 "runtime"
 "strings"
 "time"

 rotatelogs "github.com/lestrrat-go/file-rotatelogs"
 "github.com/rifflock/lfshook"
 "github.com/sirupsen/logrus"
)

func NewWriter(prefix string) io.Writer {
 w, err := rotatelogs.New(
  prefix+".%Y%m%d.json",
  // 建立软链接
  rotatelogs.WithLinkName(prefix),
  // 最多保存一星期
  rotatelogs.WithMaxAge(7*24*time.Hour),
  // 一天切割一次
  rotatelogs.WithRotationTime(24*time.Hour),
 )
 if err != nil {
  panic(err)
 }
 return w
}

func InitLog(prefix string) {
 wm := lfshook.WriterMap{
  logrus.DebugLevel: NewWriter(prefix + "_debug"),
  logrus.InfoLevel:  NewWriter(prefix + "_info"),
  logrus.WarnLevel:  NewWriter(prefix + "_warn"),
  logrus.ErrorLevel: NewWriter(prefix + "_error"),
  logrus.FatalLevel: NewWriter(prefix + "_fatal"),
 }
 logrus.AddHook(lfshook.NewHook(
  wm,
  &logrus.JSONFormatter{
   CallerPrettyfier: func(f *runtime.Frame) (string, string) {
    var callerName, fileName string
    names := strings.SplitAfterN(f.File, prefix, 2)
    if len(names) > 1 {
     fileName = fmt.Sprintf("%v;%v", names[1], f.Line)
    }

    names = strings.SplitAfterN(f.Function, prefix, 2)
    if len(names) > 1 {
     callerName = names[1]
    } else {
     callerName = f.Function
    }

    return callerName, fileName
   },
   PrettyPrint: true,
  },
 ))

 logrus.SetOutput(io.Discard)
 logrus.SetReportCaller(true)
 logrus.SetLevel(logrus.InfoLevel)

 logrus.WithFields(logrus.Fields{
  "prefix": prefix,
  "level":  logrus.GetLevel(),
 }).Warn("日志初始化完成")
}

Go 日志更新

Go 1.21 起标准库多了一个 log/slog 结构化日志模块，官方推荐用它替换老 log 包：

2026 年怎么选：新项目闭眼选 slog；logrus 生态还在但上游已近半躺平，zap 则在冲性能或要高度自定义输出时更常见。slog 的核心优势是标准库自带——框架、中间件零额外依赖就能统一接到结构化日志上。

import "log/slog"

// 基本使用
slog.Info("user logged in", "user_id", 123, "ip", "10.0.0.1")
// 输出(TextHandler 默认格式): time=2026-08-25T10:00:00.000+08:00 level=INFO msg="user logged in" user_id=123 ip=10.0.0.1

// JSON 格式
logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
logger.Info("request completed",
    "method", "GET",
    "path", "/api/users",
    "status", 200,
    "duration_ms", 15,
)

// 自定义 Handler
handler := slog.NewTextHandler(os.Stdout, &slog.HandlerOptions{
    Level: slog.LevelDebug,
    AddSource: true,
})
logger := slog.New(handler)

Go 1.21+ slog 结构化日志底层原理

slog 是 Go 1.21 引入的标准结构化日志库，设计目标是高性能、可扩展、与现有 log 包兼容。

slog 架构

┌─────────────────────────────────────────┐
│  Logger（前端 API）                       │
│  - Info/Warn/Error/Debug                 │
│  - With(name, value) 添加上下文           │
│  - Enabled(level) 检查级别               │
└──────────────┬──────────────────────────┘
               │
               ▼
┌─────────────────────────────────────────┐
│  Handler（后端接口）                       │
│  - Enabled(ctx, level) bool              │
│  - Handle(ctx, Record) error             │
│  - WithAttrs(attrs) Handler              │
│  - WithGroup(name) Handler               │
└──────────────┬──────────────────────────┘
               │
       ┌───────┴───────┐
       ▼               ▼
┌─────────────┐  ┌─────────────┐
│ TextHandler │  │ JSONHandler │
│ key=value   │  │ {"key":...} │
└─────────────┘  └─────────────┘

Record 结构（零分配优化）

// src/log/slog/record.go
type Record struct {
    Time time.Time
    Level Level
    Message string
    PC uintptr  // 程序计数器，用于获取调用位置
    
    // 关键优化：前 5 个 Attr 存在数组中，避免堆分配
    n int  // 实际 Attr 数量
    front [5]Attr  // 内联存储，栈上分配
    back []Attr  // 超过 5 个时才堆分配
}

type Attr struct {
    Key string
    Value Value  // Value 是 interface，支持任意类型
}

// 性能优化：如果日志级别不启用，直接返回，不构造 Record
func (l *Logger) Log(ctx context.Context, level Level, msg string, args ...any) {
    if !l.Enabled(ctx, level) {
        return  // 级别不启用，零开销
    }
    var pcs [1]uintptr
    runtime.Callers(3, pcs[:])  // 获取调用栈
    r := NewRecord(time.Now(), level, msg, pcs[0])
    r.Add(args...)  // 添加键值对
    _ = l.Handler().Handle(ctx, r)
}

日志框架性能对比

Benchmark 结果（100万条日志，Apple M1, Go 1.22）：

框架                  耗时        内存分配    评价
─────────────────────────────────────────────────────
slog (JSONHandler)   312 ms/op   2 allocs/op  标准库，零依赖
slog (TextHandler)   285 ms/op   1 allocs/op  文本格式更快
zap (Sugared)        198 ms/op   5 allocs/op  最快，但 API 繁琐
zap (Logger)         156 ms/op   0 allocs/op  极致性能，强类型
zerolog              178 ms/op   0 allocs/op  链式 API，零分配
logrus               890 ms/op   25 allocs/op  最慢，反射开销大
标准库 log            245 ms/op   3 allocs/op  非结构化

结论：
  - 新项目优先用 slog（标准库、性能好、可扩展）
  - 极致性能场景用 zap 或 zerolog
  - 避免用 logrus（性能差、已停止活跃维护）

生产环境最佳实践

  • 1. 全局 Logger 配置 
    func initLogger() {
    // JSON 格式，生产环境用 Info 级别
    handler := slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{
        Level: slog.LevelInfo,
        AddSource: true,  // 添加调用位置（文件:行号）
        ReplaceAttr: func(groups []string, a slog.Attr) slog.Attr {
            // 自定义字段名：time → timestamp, level → severity
            if a.Key == slog.TimeKey { return slog.Attr{Key: "timestamp", Value: a.Value} }
            if a.Key == slog.LevelKey { return slog.Attr{Key: "severity", Value: a.Value} }
            return a
        },
    })
    slog.SetDefault(slog.New(handler))
}
  • 2. 请求级上下文日志 
    func middleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // 为每个请求创建带 trace_id 的 logger
        logger := slog.Default().With(
            "trace_id", generateTraceID(),
            "method", r.Method,
            "path", r.URL.Path,
        )
        ctx := slog.NewContext(r.Context(), logger)
        next.ServeHTTP(w, r.WithContext(ctx))
    })
}

func handleUser(w http.ResponseWriter, r *http.Request) {
    logger := slog.FromContext(r.Context())
    logger.Info("处理用户请求", "user_id", 123)
    // 输出自动包含 trace_id, method, path
}
  • 3. 避免在热路径中打 Debug 日志 
    // ❌ 错误：即使 Debug 不启用，也会执行字符串拼接
logger.Debug("处理订单: " + orderID + " 金额: " + fmt.Sprintf("%d", amount))

// ✅ 正确：用 slog 的惰性求值，级别不启用时不执行
logger.Debug("处理订单", "order_id", orderID, "amount", amount)

// ✅ 更优：先检查级别
if logger.Enabled(ctx, slog.LevelDebug) {
    logger.Debug("处理订单", "order_id", orderID, "detail", computeDetail())
}
  • 4. 日志采样（高流量场景） 
    // 自定义采样 Handler，只记录 10% 的 Debug 日志
type SamplingHandler struct {
    slog.Handler
    sampleRate float64  // 0.0 - 1.0
}

func (h *SamplingHandler) Handle(ctx context.Context, r slog.Record) error {
    if r.Level == slog.LevelDebug && rand.Float64() > h.sampleRate {
        return nil  // 跳过
    }
    return h.Handler.Handle(ctx, r)
}

常见陷阱

  • 陷阱1：密码、Token 等敏感信息写入日志 → 使用自定义 ReplaceAttr 脱敏
  • 陷阱2：日志中拼接用户输入导致日志注入 → 用结构化字段，不要拼接字符串
  • 陷阱3：在循环中打 Info 日志 → 高 QPS 下会打爆磁盘，用采样或聚合
  • 陷阱4：忘记 defer cancel() 导致 context 泄漏 → 见 context 章节的内存泄漏部分
