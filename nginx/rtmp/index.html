RTMP 模块更新

RTMP 模块现状与替代方案：

  • Nginx-rtmp-module：已多年未更新（最后更新 2021），存在兼容性问题
  • SRT (Secure Reliable Transport)：IETF 标准，安全性更高
  • WebRTC：低延迟直播，更好的浏览器兼容
  • HLS + CMAF：Apple 和 Microsoft 推动的标准化

RTMP 配置

基础配置

rtmp {
    server {
        listen 1935;        # RTMP 端口
        chunk_size 4096;

        application live {
            live on;
            record off;

            # 录制配置
            # record *;
            # record_path /tmp/recordings;
            # record_unique on;

            # HLS 配置
            hls on;
            hls_path /tmp/hls;
            hls_fragment 3;
            hls_playlist_length 60;
            hls_type event;
            wait_video on;
            # publish_notify on;
            # drop_idle_publisher 30s;
        }
    }
}

http {
    server {
        listen 8080;

        location /hls {
            types {
                application/vnd.apple.mpegurl m3u8;
                video/mp2t ts;
            }
            root /tmp;
            add_header Cache-Control no-cache;
            add_header Access-Control-Allow-Origin *;
        }
    }
}

推流与拉流

rtmp {
    server {
        listen 1935;

        application live {
            live on;

            # 推流密钥保护
            on_publish http://localhost:8080/auth;
            on_publish_done http://localhost:8080/auth_done;

            # 允许从本机推流
            allow publish 127.0.0.1;
            allow publish 192.168.0.0/16;
            deny publish all;

            # 允许所有客户端观看
            allow play all;

            # 转推其他服务器
            push rtmp://other-server:1935/live/stream_key;

            # 录制当前直播
            record all;
            record_path /tmp/recordings;
            record_suffix -%Y%m%d-%H%M%S.flv;
        }

        # 多级联播
        application backup {
            live on;
            play /tmp/recordings;
        }
    }
}

# 视频设置
编码器: x264 (或硬件编码)
比特率: 6000 kbps

采样率: 44.1 kHz

  • 硬件编码：NVIDIA NVENC / AMD VCE / Apple VT
  • 编码器预设：quality / speed / veryfast
  • Profile：high / main / baseline
  • -tune：zerolatency / fastdecode

FFmpeg 推流

基础推流

# 推送本地视频文件
ffmpeg -re -i input.mp4 -c:v libx264 -c:a aac     -f flv rtmp://localhost:1935/live/stream1

# 推送屏幕录制
ffmpeg -f avfoundation -i "1:0" -c:v libx264 -c:a aac     -f flv rtmp://localhost:1935/live/stream1

# 推送摄像头
ffmpeg -f avfoundation -i "0:0" -c:v libx264 -preset veryfast     -b:v 4000k -c:a aac -b:a 128k     -f flv rtmp://localhost:1935/live/stream1

# 推流同时录制
ffmpeg -re -i input.mp4 -c:v libx264 -c:a aac     -f flv rtmp://localhost:1935/live/stream1     -c:v libx264 -c:a aac -f mp4 /tmp/recording.mp4

转码推流

# 多种画质同时推流
ffmpeg -re -i input.mp4     -c:v libx264 -b:v 6000k -c:a aac -b:a 192k -f flv rtmp://localhost:1935/live/hd     -c:v libx264 -b:v 3000k -c:a aac -b:a 128k -f flv rtmp://localhost:1935/live/sd     -c:v libx264 -b:v 1500k -c:a aac -b:a 96k -f flv rtmp://localhost:1935/live/mobile

# 硬件编码推流 (macOS)
ffmpeg -re -i input.mp4 -c:v h264_videotoolbox -c:a aac     -f flv rtmp://localhost:1935/live/stream1

# 硬件编码推流 (NVIDIA)
ffmpeg -re -i input.mp4 -c:v h264_nvenc -c:a aac     -f flv rtmp://localhost:1935/live/stream1

HLS 输出

# 输出 HLS
ffmpeg -re -i input.mp4 -c:v libx264 -c:a aac     -f hls -hls_time 4 -hls_list_size 10     -hls_segment_filename /tmp/hls/%03d.ts     /tmp/hls/index.m3u8

# 纯音频 HLS
ffmpeg -re -i input.mp4 -c:a aac -b:a 128k     -f hls -hls_time 10 -hls_list_size 10     -hls_segment_filename /tmp/hls/audio_%03d.aac     /tmp/hls/audio.m3u8

监控与统计

rtmp_stat 模块

http {
    server {
        location /stat {
            rtmp_stat all;
            rtmp_stat_stylesheet stat.xsl;
        }

        location /stat.xsl {
            root /path/to/nginx-rtmp-module/;
        }
    }
}

FFprobe 分析

# 查看流信息
ffprobe -v quiet -print_format json -show_streams rtmp://localhost:1935/live/stream1

# 查看实时统计
ffprobe -v error -show_entries stream=bit_rate,codec_name rtmp://localhost:1935/live/stream1

# 录制流
ffmpeg -i rtmp://localhost:1935/live/stream1 -c copy output.mp4

关键指标

  • 延迟：GOP 大小、关键帧间隔
  • 码率：视频 + 音频比特率
  • 帧率：实际编码帧率
  • 丢帧：网络拥塞检测
  • 连接数：并发观看人数
